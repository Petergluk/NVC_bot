def _load_prompt(file_name: str) -> str:
    """
    Загружает промпт из файла, игнорируя строки-комментарии.
    Если файл не найден или пуст, возвращает пустую строку.
    
    Args:
        file_name (str): Имя файла в директории /prompts.
        
    Returns:
        str: "Чистое" содержимое промпта без комментариев или пустая строка.
    """
    file_path = PROMPTS_DIR / file_name
    if not file_path.exists():
        logging.warning("Файл промпта не найден: %s. Будет использовано пустое значение.", file_path)
        return ""

    try:
        # Читаем все строки из файла
        raw_lines = file_path.read_text(encoding='utf-8').splitlines()
        # Фильтруем строки: оставляем только те, которые не начинаются с '#' (после удаления пробелов слева)
        prompt_lines = [line for line in raw_lines if not line.strip().startswith('#')]
        # Соединяем отфильтрованные строки обратно в единый текст
        return "\n".join(prompt_lines).strip()
    except IOError as e:
        logging.error("Не удалось прочитать файл промпта %s: %s. Будет использовано пустое значение.", file_path, e)
        return ""

# bot/database.py
# NVC_bot/bot/database.py
# version: 2.0.1
# 2025-07-29T15:20:00Z
"""
Модуль для асинхронной работы с базой данных SQLite.

Version 2.0.1:
- FIX: В методе `get_chat_history` убрано добавление имени пользователя в текст сообщения.
       Это соответствует новому, более чистому подходу к формированию истории,
       где имя пользователя добавляется в `user.py` только для последнего сообщения.

Version 2.0.0:
- REFACTOR: Метод `get_chat_history` полностью переработан.
            - Теперь он возвращает список словарей (`List[Dict]`) вместо строки.
            - Запрос к БД теперь использует `ORDER BY timestamp ASC` для прямого
              получения сообщений в хронологическом порядке.
"""
import logging
import json
from typing import Any, Dict, List, Optional, Type, TypeVar
from collections import defaultdict

import aiosqlite

from .config import DATA_DIR, settings

logger = logging.getLogger(__name__)
T = TypeVar('T')

class Database:
    """
    Класс для управления базой данных SQLite.
    """

    def __init__(self):
        self.db_path = DATA_DIR / "bot_database.sqlite3"
        self.conn: Optional[aiosqlite.Connection] = None
        logger.info("Объект Database создан. Путь к БД: %s", self.db_path)

    async def initialize(self):
        try:
            self.conn = await aiosqlite.connect(self.db_path)
            self.conn.row_factory = aiosqlite.Row
            await self._create_tables()
            logger.info("Подключение к базе данных установлено и таблицы проверены.")
        except aiosqlite.Error as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА при инициализации БД: %s", e, exc_info=True)
            raise

    async def _create_tables(self):
        if not self.conn: raise RuntimeError("Подключение к БД не инициализировано.")
        async with self.conn.cursor() as cursor:
            await cursor.execute("CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT NOT NULL)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS messages (id INTEGER PRIMARY KEY AUTOINCREMENT, chat_id INTEGER NOT NULL, user_name TEXT NOT NULL, text TEXT NOT NULL, is_bot BOOLEAN NOT NULL, timestamp DATETIME DEFAULT CURRENT_TIMESTAMP)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS statistics (key TEXT PRIMARY KEY, value INTEGER NOT NULL)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS user_warnings (user_id INTEGER PRIMARY KEY, count INTEGER NOT NULL DEFAULT 0)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS moderation_triggers (id INTEGER PRIMARY KEY AUTOINCREMENT, trigger_word TEXT NOT NULL UNIQUE, category TEXT NOT NULL DEFAULT 'general')")
            await cursor.execute("CREATE TABLE IF NOT EXISTS moderation_exceptions (id INTEGER PRIMARY KEY AUTOINCREMENT, exception_phrase TEXT NOT NULL UNIQUE, category TEXT NOT NULL DEFAULT 'general')")
            
            await cursor.execute("PRAGMA table_info(moderation_triggers)")
            if 'category' not in [row['name'] for row in await cursor.fetchall()]:
                await cursor.execute("ALTER TABLE moderation_triggers ADD COLUMN category TEXT NOT NULL DEFAULT 'general'")
            
            await cursor.execute("PRAGMA table_info(moderation_exceptions)")
            if 'category' not in [row['name'] for row in await cursor.fetchall()]:
                await cursor.execute("ALTER TABLE moderation_exceptions ADD COLUMN category TEXT NOT NULL DEFAULT 'general'")
        await self.conn.commit()

    async def populate_defaults(self):
        if not self.conn: return
        async with self.conn.execute("SELECT COUNT(*) FROM moderation_triggers") as cursor:
            if (await cursor.fetchone())[0] == 0:
                logger.info("Таблица триггеров пуста. Заполняю...")
                await self.reset_triggers()
        async with self.conn.execute("SELECT COUNT(*) FROM moderation_exceptions") as cursor:
            if (await cursor.fetchone())[0] == 0:
                logger.info("Таблица исключений пуста. Заполняю...")
                await self.reset_exceptions()

    async def get_all_triggers(self) -> Dict[str, List[str]]:
        if not self.conn: return {}
        triggers = defaultdict(list)
        try:
            async with self.conn.execute("SELECT trigger_word, category FROM moderation_triggers ORDER BY category, id") as cursor:
                async for row in cursor:
                    triggers[row['category']].append(row['trigger_word'])
        except aiosqlite.Error as e: logger.error("Ошибка БД при получении триггеров: %s", e)
        return dict(triggers)

    async def get_all_triggers_flat(self) -> List[str]:
        if not self.conn: return []
        try:
            async with self.conn.execute("SELECT trigger_word FROM moderation_triggers") as cursor:
                return [row['trigger_word'] for row in await cursor.fetchall()]
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при получении плоского списка триггеров: %s", e)
            return []

    async def add_trigger(self, trigger: str, category: str = 'general') -> bool:
        if not self.conn: return False
        try:
            await self.conn.execute("INSERT OR IGNORE INTO moderation_triggers (trigger_word, category) VALUES (?, ?)", (trigger.lower(), category.lower()))
            await self.conn.commit()
            return self.conn.total_changes > 0
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при добавлении триггера '%s': %s", trigger, e)
            return False

    async def delete_trigger(self, trigger: str) -> bool:
        if not self.conn: return False
        try:
            await self.conn.execute("DELETE FROM moderation_triggers WHERE trigger_word = ?", (trigger.lower(),))
            await self.conn.commit()
            return self.conn.total_changes > 0
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при удалении триггера '%s': %s", trigger, e)
            return False

    async def reset_triggers(self):
        if not self.conn: return
        try:
            async with self.conn.cursor() as cursor:
                await cursor.execute("DELETE FROM moderation_triggers")
                data_to_insert = [(word.lower(), category.lower()) for category, words in settings.DEFAULT_TRIGGERS.items() for word in words]
                await cursor.executemany("INSERT OR IGNORE INTO moderation_triggers (trigger_word, category) VALUES (?, ?)", data_to_insert)
            await self.conn.commit()
            logger.info("Таблица триггеров успешно сброшена к значениям по умолчанию.")
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при сбросе триггеров: %s", e)
            await self.conn.rollback()

    async def get_all_exceptions(self) -> Dict[str, List[str]]:
        if not self.conn: return {}
        exceptions = defaultdict(list)
        try:
            async with self.conn.execute("SELECT exception_phrase, category FROM moderation_exceptions ORDER BY category, id") as cursor:
                async for row in cursor:
                    exceptions[row['category']].append(row['exception_phrase'])
        except aiosqlite.Error as e: logger.error("Ошибка БД при получении исключений: %s", e)
        return dict(exceptions)
    
    async def get_all_exceptions_flat(self) -> List[str]:
        if not self.conn: return []
        try:
            async with self.conn.execute("SELECT exception_phrase FROM moderation_exceptions") as cursor:
                return [row['exception_phrase'] for row in await cursor.fetchall()]
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при получении плоского списка исключений: %s", e)
            return []

    async def add_exception(self, exception: str, category: str = 'general') -> bool:
        if not self.conn: return False
        try:
            await self.conn.execute("INSERT OR IGNORE INTO moderation_exceptions (exception_phrase, category) VALUES (?, ?)", (exception.lower(), category.lower()))
            await self.conn.commit()
            return self.conn.total_changes > 0
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при добавлении исключения '%s': %s", exception, e)
            return False

    async def delete_exception(self, exception: str) -> bool:
        if not self.conn: return False
        try:
            await self.conn.execute("DELETE FROM moderation_exceptions WHERE exception_phrase = ?", (exception.lower(),))
            await self.conn.commit()
            return self.conn.total_changes > 0
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при удалении исключения '%s': %s", exception, e)
            return False

    async def reset_exceptions(self):
        if not self.conn: return
        try:
            async with self.conn.cursor() as cursor:
                await cursor.execute("DELETE FROM moderation_exceptions")
                data_to_insert = [(phrase.lower(), category.lower()) for category, phrases in settings.DEFAULT_EXCEPTIONS.items() for phrase in phrases]
                await cursor.executemany("INSERT OR IGNORE INTO moderation_exceptions (exception_phrase, category) VALUES (?, ?)", data_to_insert)
            await self.conn.commit()

class AdminHandlers:

        if update.message:
            try: await update.message.delete()

        await self.callback_handlers.show_facilitator_settings_menu(update, context)
        return FACILITATOR_SETTINGS

    async def moderate_entry_point(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        if not self.callback_handlers: return ConversationHandler.END
        if update.message:
            try: await update.message.delete()

        await self.callback_handlers.show_moderate_settings_menu(update, context)
        return MODERATION_SETTINGS

    async def _commit_prompt_changes(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        trace_id = context.user_data.get('trace_id', str(uuid.uuid4()))
        prompt_type = context.user_data.get("prompt_type")
        new_text = context.user_data.get("prompt_buffer", "")
        original_update = context.user_data.get("original_update")

        if not all([prompt_type, new_text, original_update]):
            logger.error("Недостаточно данных для сохранения промпта.", extra={'trace_id': trace_id})
            clear_conversation_user_data(context)
            return

        original_query = original_update.callback_query
        success, error_message = False, ""
        
        save_actions = {
            "facilitator_prompt": (validate_prompt, self.db.set_prompt),
            "moderation_prompt": (lambda p: None if "{text}" in p else "Промпт должен содержать `{text}`.", self.db.set_moderation_prompt),
            "moderation_template": (validate_moderation_template, self.db.set_moderation_task_template),
            "welcome_prompt": (lambda p: None if "{user_mention}" in p else "Промпт должен содержать `{user_mention}`.", self.db.set_welcome_prompt)
        }

        validator, saver = save_actions.get(prompt_type, (lambda p: "Неизвестный тип промпта", None))
        error_message = validator(new_text)
        if not error_message and saver:
            await saver(new_text)
            success = True
            logger.info("Промпт '%s' успешно сохранен. Длина: %d", prompt_type, len(new_text), extra={'trace_id': trace_id})

        if success:
            await context.bot.send_message(chat_id=original_query.message.chat_id, text=f"✅ Промпт успешно обновлен\\. Длина: {len(new_text)}\\.")
        else:
            await context.bot.send_message(chat_id=original_query.message.chat_id, text=f"❌ Ошибка валидации: {escape_markdown_v2(error_message)}", parse_mode=ParseMode.MARKDOWN_V2)
        
        # FIX: Используем безопасную очистку
        clear_conversation_user_data(context)
        
        menu_update_map = {
            "facilitator_prompt": self.callback_handlers.show_facilitator_prompt_menu,
            "moderation_prompt": self.callback_handlers.show_moderation_prompt_menu,
            "moderation_template": self.callback_handlers.show_moderation_template_menu,
            "welcome_prompt": self.callback_handlers.show_welcome_prompt_menu,
        }
        if update_func := menu_update_map.get(prompt_type):
            await update_func(original_update, context)

    @admin_only
    @log_action("handle_prompt_message")
    async def handle_prompt_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        if 'prompt_commit_task' in context.user_data:
            context.user_data['prompt_commit_task'].cancel()

        buffer = context.user_data.get("prompt_buffer", "")
        context.user_data["prompt_buffer"] = buffer + update.message.text + "\n"
        
        commit_task = asyncio.create_task(self._commit_prompt_changes(update, context))
        context.user_data['prompt_commit_task'] = commit_task

        return AWAIT_PROMPT_INPUT

    @admin_only
    @log_action("cancel_conversation")
    async def cancel_conversation(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        if 'prompt_commit_task' in context.user_data:
            context.user_data['prompt_commit_task'].cancel()
        
        query = update.callback_query
        if query and query.message:
            try: await query.delete_message()
            except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить сообщение меню при отмене: %s", e)
        elif update.message:
            original_update = context.user_data.get("original_update")
            try: await update.message.delete()
            except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить команду /cancel: %s", e)
            if original_update and self.callback_handlers:
                if original_update.callback_query:
                    await original_update.callback_query.delete_message()

        # FIX: Используем безопасную очистку
        clear_conversation_user_data(context)
        return ConversationHandler.END

    @admin_only
    @log_action("warnings")
    async def manage_warnings(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        args, message, trace_id = context.args, update.message, str(uuid.uuid4())
        target_id, amount_str = None, None
        
        if message.reply_to_message:
            target_id = message.reply_to_message.from_user.id
            if args: amount_str = args[0]
        elif len(args) >= 1:
            target_id_str = args[0]
            if len(args) > 1 and re.match(r"^[+-]?\d+$", args[1]): amount_str = args[1]
            try:
                if target_id_str.isdigit(): target_id = int(target_id_str)
                elif target_id_str.startswith('@'): target_id = (await context.bot.get_chat(chat_id=target_id_str)).id
                else: raise ValueError("Неверный формат")
            except (TelegramError, ValueError):
                await message.reply_text(f"❌ Не удалось найти пользователя `{escape_markdown_v2(target_id_str)}`\\.", parse_mode=ParseMode.MARKDOWN_V2)
                return
        
        if not target_id:
            await message.reply_text("*Как использовать:* `/warnings <user_id/@username> [+/-N]` или ответом на сообщение\\.", parse_mode=ParseMode.MARKDOWN_V2)
            return

        try:
            user_info = await context.bot.get_chat(target_id)
            user_mention = escape_markdown_v2(f"@{user_info.username}" if user_info.username else user_info.full_name)
        except Exception as e:
            logger.warning("Не удалось получить инфо о пользователе %d: %s", target_id, e, extra={'trace_id': trace_id})
            user_mention = f"пользователь с ID `{target_id}`"

        if amount_str:
            if not re.match(r"^[+-]?\d+$", amount_str):
                await message.reply_text("❌ Неверный формат количества: `+1`, `\\-1`\\.", parse_mode=ParseMode.MARKDOWN_V2)
                return
            new_count = await self.db.adjust_warnings(target_id, int(amount_str))
            await message.reply_text(f"✅ Варны для {user_mention} изменены\\. Теперь: *{new_count}*\\.", parse_mode=ParseMode.MARKDOWN_V2)
        else:
            count = await self.db.get_warnings(target_id)
            await message.reply_text(f"ℹ️ У {user_mention} *{count}* варн\\(ов\\)\\.", parse_mode=ParseMode.MARKDOWN_V2)

        Инициализация базового обработчика.

        Args:
            db (Database): Экземпляр для работы с базой данных.
            gemini (GeminiService): Экземпляр для работы с Gemini API.
        """
        self.db = db
        self.gemini = gemini

    async def _safe_edit_message(self, query: CallbackQuery, text: str, **kw: Any) -> Message | None:
        """Безопасно редактирует сообщение, на которое нажал пользователь."""
        if not query or not query.message:
            logger.warning("Попытка редактировать сообщение без query или query.message.")
            return None
        try:
            # Мы не обрезаем текст здесь, чтобы иметь возможность отловить ошибку о длине
            return await query.message.edit_text(text, **kw)
        except BadRequest as exc:
            # FIX: Детальная обработка исключений BadRequest
            if "Message is too long" in str(exc):
                logger.warning("Не удалось отредактировать сообщение: превышен лимит длины (4096 символов).")
                await self._safe_answer_query(query, "❌ Ошибка: Слишком много текста для отображения.", show_alert=True)
            elif "Message is not modified" in str(exc):
                # Это не ошибка, просто игнорируем
                await self._safe_answer_query(query)
            else:
                logger.error("Telegram BadRequest при редактировании сообщения: %s", exc)
                await self._safe_answer_query(query, "Произошла ошибка при обновлении меню.", show_alert=True)
            return None

    async def _safe_answer_query(self, query: CallbackQuery | None, *args, **kwargs) -> None:
        """Безопасно отвечает на callback query, игнорируя устаревшие."""
        if not query: return
        try:
            await query.answer(*args, **kwargs)
        except BadRequest as e:
            if "Query is too old" in str(e):
                logger.warning("Попытка ответить на устаревший query. Игнорируется.")
            else:
                logger.error("Неожиданная ошибка BadRequest при ответе на query: %s", e)
    
    async def _disable_keyboard(self, query: CallbackQuery) -> None:
        """Безопасно удаляет клавиатуру у сообщения."""
        if not query or not query.message: return
        try:
            await query.message.edit_reply_markup(reply_markup=None)
        except BadRequest as e:
            if "Message is not modified" not in str(e):
                logger.warning("Не удалось убрать клавиатуру: %s", e)

"""
Обработчики инлайн-кнопок для меню настроек параметров контекста AI.

Version 1.0.2:
- FIX: Все сообщения переведены на `ParseMode.MARKDOWN_V2`.
- FIX: Все статические и динамические данные теперь корректно экранируются.
"""
import logging
from typing import Dict
from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode
from telegram.ext import ContextTypes
from telegram.error import BadRequest, TelegramError

from ...utils.formatters import escape_markdown_v2
from ..states import FACILITATOR_SETTINGS, CONTEXT_SETTINGS, AWAIT_CONTEXT_VALUE
from .callback_patterns import CallbackPrefix

logger = logging.getLogger(__name__)

class ContextCallbackHandler:
    """Обработчики для ветки /settings -> Настройки контекста."""

    CONTEXT_PARAM_META = {
        "context_messages_limit": {"name": "Контекст диалога", "type": int, "range": (0, 100)},
        "context_judge_limit": {"name": "Контекст для Судьи", "type": int, "range": (0, 50)},
    }

    async def show_context_settings_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Показывает меню настроек лимитов контекста."""
        query = update.callback_query
        if query:
            await self._safe_answer_query(query)

        config = await self.db.get_context_config()
        
        text_lines = [
            "📚 *Настройки Контекста*\n",
            "Здесь настраивается, сколько последних сообщений бот 'помнит' при ответе или модерации\\.",
            "• *Контекст диалога:* Глубина памяти для обычных ответов\\.",
            "• *Контекст для Судьи:* Глубина памяти для анализа нарушений\\.\n"
        ]
        
        buttons = []
        for key, meta in self.CONTEXT_PARAM_META.items():
            value = config.get(key)
            text_lines.append(f"*{escape_markdown_v2(meta['name'])}*: `{value}` сообщ\\.")
            buttons.append(InlineKeyboardButton(f"{meta['name']}", callback_data=f"{CallbackPrefix.CONTEXT_CONFIG_EDIT}{key}"))
        
        kb = [buttons[i:i + 2] for i in range(0, len(buttons), 2)]
        kb.append([InlineKeyboardButton("🔄 Сбросить", callback_data=CallbackPrefix.CONTEXT_CONFIG_RESET), InlineKeyboardButton("⬅️ Назад", callback_data=CallbackPrefix.BACK_TO_FACILITATOR_SETTINGS)])
        
        text = "\n".join(text_lines)
        reply_markup = InlineKeyboardMarkup(kb)
        
        await self._safe_edit_message(query, text, reply_markup=reply_markup, parse_mode=ParseMode.MARKDOWN_V2)
        return CONTEXT_SETTINGS

    async def request_context_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает новое значение для параметра контекста."""
        query = update.callback_query
        await self._safe_answer_query(query)

        param_key = query.data.replace(CallbackPrefix.CONTEXT_CONFIG_EDIT, "")
        if param_key not in self.CONTEXT_PARAM_META:
            return CONTEXT_SETTINGS

        meta = self.CONTEXT_PARAM_META[param_key]
        context.user_data["context_param_key"] = param_key
        context.user_data["original_update"] = update
        
        current_value = (await self.db.get_context_config()).get(param_key)
        min_val, max_val = meta["range"]

        text = (
            f"✏️ *Изменение параметра: {escape_markdown_v2(meta['name'])}*\n\n"
            f"Этот параметр определяет максимальное количество последних сообщений из истории чата, которое будет использоваться\\.\n\n"
            f"*Текущее значение: `{current_value}`*\n"
            f"Введите целое число от `{min_val}` до `{max_val}`\\.\n\n"
            "**Отправьте новое значение или /cancel для отмены\\.**"
        )
        
        await self._safe_edit_message(query, text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        return AWAIT_CONTEXT_VALUE

    async def set_context_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Устанавливает новое значение для параметра контекста."""
        msg = update.message
        param_key = context.user_data.get("context_param_key")

        if not param_key or param_key not in self.CONTEXT_PARAM_META:
            await msg.reply_text("❌ Внутренняя ошибка: не удалось определить параметр для изменения\\.")
            context.user_data.clear()
            return FACILITATOR_SETTINGS

        meta = self.CONTEXT_PARAM_META[param_key]
        value_str = msg.text.strip()
        
        try:
            value = meta["type"](value_str)
            min_val, max_val = meta["range"]
            if not (min_val <= value <= max_val):
                raise ValueError(f"Значение должно быть в диапазоне от {min_val} до {max_val}\\.")
        except (ValueError, TypeError) as e:
            await msg.delete()
            original_update = context.user_data.get("original_update")
            if original_update and original_update.callback_query:
                error_text = (
                    f"❌ *Неверное значение для {escape_markdown_v2(meta['name'])}*\n\n"
                    f"Пожалуйста, введите *целое число* от `{meta['range'][0]}` до `{meta['range'][1]}`\\.\n"
                    f"Ошибка: {escape_markdown_v2(str(e))}\n\n"
                    f"**Попробуйте снова или нажмите /cancel\\.**"
                )
                await self._safe_edit_message(original_update.callback_query, error_text, parse_mode=ParseMode.MARKDOWN_V2)
            return AWAIT_CONTEXT_VALUE

        await self.db.set_context_config_value(param_key, value)
        
        try:
            await msg.delete()
        except (BadRequest, TelegramError) as e:
            logger.warning("Не удалось удалить сообщение пользователя при изменении параметра контекста: %s", e)

        original_update = context.user_data.pop("original_update", None)
        context.user_data.clear()
        
        if original_update:
            await self._safe_answer_query(original_update.callback_query, f"✅ {meta['name']} обновлен.")
            return await self.show_context_settings_menu(original_update, context)
        
        return FACILITATOR_SETTINGS

    async def reset_context_config_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Сбрасывает параметры контекста к значениям по умолчанию."""
        query = update.callback_query
        await self._safe_answer_query(query, "Выполняется сброс...")
        
        await self.db.reset_context_config()
        
        await self._safe_answer_query(query, "✅ Параметры контекста сброшены.")
        return await self.show_context_settings_menu(update, context)

    async def set_generation_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        msg = update.message
        param_key = context.user_data.get("gen_param_key")

        if not param_key or param_key not in self.GENERATION_PARAM_META:
            await msg.reply_text("❌ Внутренняя ошибка: не удалось определить параметр для изменения\\.")
            context.user_data.clear()
            return FACILITATOR_SETTINGS

        meta = self.GENERATION_PARAM_META[param_key]
        value_str = msg.text.strip().replace(',', '.')
        
        try:
            value = meta["type"](value_str)
            min_val, max_val = meta["range"]
            if min_val is not None and value < min_val:
                raise ValueError(f"значение должно быть не меньше {min_val}")
            if max_val is not None and value > max_val:
                raise ValueError(f"значение должно быть не больше {max_val}")
        except (ValueError, TypeError) as e:
            await msg.delete()
            original_update = context.user_data.get("original_update")
            if original_update and original_update.callback_query:
                error_text = (
                    f"❌ *Неверное значение для {escape_markdown_v2(meta['name'])}*\n\n"
                    f"Пожалуйста, введите *целое число* от `{meta['range'][0]}` до `{meta['range'][1]}`\\.\n"
                    f"Ошибка: {escape_markdown_v2(str(e))}\n\n"
                    f"**Попробуйте снова или нажмите /cancel\\.**"
                )
                await self._safe_edit_message(original_update.callback_query, error_text, parse_mode=ParseMode.MARKDOWN_V2)
            return AWAIT_GENERATION_VALUE

        await self.db.set_generation_config_value(param_key, value)
        
        try:
            await msg.delete()
        except (BadRequest, TelegramError) as e:
            logger.warning("Не удалось удалить сообщение пользователя при изменении параметра: %s", e)

        original_update = context.user_data.pop("original_update", None)
        context.user_data.clear()
        
        if original_update:
            await self._safe_answer_query(original_update.callback_query, f"✅ {meta['name']} обновлен.")
            return await self.show_generation_settings_menu(original_update, context)
        
        return FACILITATOR_SETTINGS

    async def reset_generation_config_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        await self._safe_answer_query(query, "Выполняется сброс...")
        
        await self.db.reset_generation_config()
        
        await self._safe_answer_query(query, "✅ Параметры сброшены.")
        return await self.show_generation_settings_menu(update, context)

"""
Обработчики инлайн-кнопок для управления триггерами и исключениями модерации.

Version 1.8.0:
- REFACTOR: Методы `request_items_add` и `request_items_delete` были переработаны
            для устранения дублирования кода. Введен новый приватный метод
            `_request_item_modification_input`, который централизует логику
            запроса ввода от пользователя.
Version 1.7.0:
- REFACTOR: Изменена логика `_format_list_for_display` для решения проблемы
            с лимитом на количество "сущностей" (entities) в Telegram.
- FIX: Вместо оборачивания каждого слова в `<code>`, теперь вся строка
       слов для одной категории оборачивается в один тег `<code>`. Это
       радикально снижает количество сущностей и предотвращает сбой форматирования.
"""
from typing import Dict, List

from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode
from telegram.ext import ContextTypes
from telegram.error import BadRequest, TelegramError
import logging

from ...utils.formatters import escape_markdown_v2, escape_html
from ..states import (MANAGE_TRIGGERS, AWAIT_TRIGGERS_TO_ADD, AWAIT_TRIGGERS_TO_DELETE,
                      MANAGE_EXCEPTIONS, AWAIT_EXCEPTIONS_TO_ADD, AWAIT_EXCEPTIONS_TO_DELETE)
from .callback_patterns import CallbackPrefix

logger = logging.getLogger(__name__)

class ItemCallbackHandler:
    """Обработчики для управления списками триггеров и исключений."""

    def _format_list_for_display(self, items: Dict[str, List[str]]) -> str:
        """
        Безопасно форматирует список для отображения в HTML.
        """
        if not items:
            return "Список пуст."
        lines = []
        for category, words in items.items():
            escaped_category = escape_html(category.capitalize())
            lines.append(f"<i>{escaped_category}</i>:")
            
            all_words_string = ", ".join(words)
            escaped_words = escape_html(all_words_string)
            lines.append(f"<code>{escaped_words}</code>")
            
        return "\n\n".join(lines)

    async def _show_item_management_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        query = update.callback_query
        await self._safe_answer_query(query)
        
        items, return_state, callbacks = {}, 0, {}
        if item_type == "triggers":
            items = await self.db.get_all_triggers()
            return_state = MANAGE_TRIGGERS
            callbacks = {"title": "🎯 <b>Управление триггерами</b>", "back": CallbackPrefix.BACK_TO_MODERATION_SETTINGS, "add": CallbackPrefix.TRIGGERS_ADD, "del": CallbackPrefix.TRIGGERS_DELETE, "reset": CallbackPrefix.TRIGGERS_RESET}
        elif item_type == "exceptions":
            items = await self.db.get_all_exceptions()
            return_state = MANAGE_EXCEPTIONS
            callbacks = {"title": "🛡️ <b>Управление исключениями</b>", "back": CallbackPrefix.BACK_TO_MODERATION_SETTINGS, "add": CallbackPrefix.EXCEPTIONS_ADD, "del": CallbackPrefix.EXCEPTIONS_DELETE, "reset": CallbackPrefix.EXCEPTIONS_RESET}
        
        text = f"{callbacks['title']}\n\n{self._format_list_for_display(items)}\n\nДля отмены введите /cancel."
        kb = [
            [InlineKeyboardButton("➕ Добавить", callback_data=callbacks["add"]), InlineKeyboardButton("➖ Удалить", callback_data=callbacks["del"])],
            [InlineKeyboardButton("🔄 Сбросить", callback_data=callbacks["reset"]), InlineKeyboardButton("⬅️ Назад", callback_data=callbacks["back"])]
        ]
        await self._safe_edit_message(query, text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.HTML)
        return return_state

    async def show_triggers_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_item_management_menu(update, context, "triggers")

    async def show_exceptions_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_item_management_menu(update, context, "exceptions")

    async def _request_item_modification_input(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str, action: str) -> int:
        """
        Универсальный метод для запроса ввода от пользователя при добавлении или удалении элементов.

        Args:
            update (Update): Объект обновления Telegram.
            context (ContextTypes.DEFAULT_TYPE): Контекст бота.
            item_type (str): Тип элемента ('triggers' или 'exceptions').
            action (str): Действие ('add' или 'delete').

        Returns:
            int: Следующее состояние для ConversationHandler.
        """
        query = update.callback_query
        await self._safe_answer_query(query)

        # Словарь для хранения текстов и состояний, чтобы избежать дублирования if/else
        config = {
            "triggers": {
                "add": ("Отправьте триггеры для добавления", AWAIT_TRIGGERS_TO_ADD),
                "delete": ("Отправьте триггеры для удаления", AWAIT_TRIGGERS_TO_DELETE),
            },
            "exceptions": {
                "add": ("Отправьте фразы-исключения для добавления", AWAIT_EXCEPTIONS_TO_ADD),
                "delete": ("Отправьте фразы-исключения для удаления", AWAIT_EXCEPTIONS_TO_DELETE),
            }
        }

        base_text, return_state = config[item_type][action]

        # Добавляем детали к базовому тексту
        if action == 'add':
            full_text = f"{base_text}\\.\nФормат: `Категория: слово1, слово2`\nИли просто `слово1, слово2` для категории 'general'\\."
        else:  # 'delete'
            full_text = f"{base_text}, перечислив их через запятую\\."

        context.user_data['original_update'] = update
        await self._safe_edit_message(query, full_text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        
        return return_state

    async def request_items_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает у пользователя элементы (триггеры/исключения) для добавления."""
        item_type = "triggers" if update.callback_query.data == CallbackPrefix.TRIGGERS_ADD else "exceptions"
        return await self._request_item_modification_input(update, context, item_type, "add")

    async def request_items_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает у пользователя элементы (триггеры/исключения) для удаления."""
        item_type = "triggers" if update.callback_query.data == CallbackPrefix.TRIGGERS_DELETE else "exceptions"
        return await self._request_item_modification_input(update, context, item_type, "delete")

    async def _handle_add_logic(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        msg = update.message
        text = msg.text.strip()
        category = 'general'
        if ':' in text:
            category, text = (part.strip() for part in text.split(':', 1))
            
        items = [item.strip().lower() for item in text.split(',') if item.strip()]
        added_count = 0
        
        add_func = self.db.add_trigger if item_type == "triggers" else self.db.add_exception
        for item in items:
            if await add_func(item, category): added_count += 1
        
        try: await msg.delete()
        except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить сообщение пользователя: %s", e)

        original_update = context.user_data.pop('original_update', None)
        if original_update and original_update.callback_query:
            await self._safe_answer_query(original_update.callback_query, text=f"✅ Добавлено {added_count} из {len(items)} в '{category}'.")
            await self._show_item_management_menu(original_update, context, item_type)
            
        return MANAGE_TRIGGERS if item_type == "triggers" else MANAGE_EXCEPTIONS

    async def _handle_delete_logic(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        msg = update.message
        items_to_delete = [item.strip().lower() for item in msg.text.strip().split(',') if item.strip()]
        deleted_count = 0
        
        delete_func = self.db.delete_trigger if item_type == "triggers" else self.db.delete_exception
        for item in items_to_delete:
            if await delete_func(item): deleted_count += 1

        try: await msg.delete()
        except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить сообщение пользователя: %s", e)

        original_update = context.user_data.pop('original_update', None)
        if original_update and original_update.callback_query:
            await self._safe_answer_query(original_update.callback_query, text=f"✅ Удалено {deleted_count} из {len(items_to_delete)} элементов.")
            await self._show_item_management_menu(original_update, context, item_type)
            
        return MANAGE_TRIGGERS if item_type == "triggers" else MANAGE_EXCEPTIONS

    async def handle_triggers_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_add_logic(update, context, "triggers")

    async def handle_triggers_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_delete_logic(update, context, "triggers")

    async def handle_exceptions_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_add_logic(update, context, "exceptions")

    async def handle_exceptions_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_delete_logic(update, context, "exceptions")

    async def reset_items(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        item_type = "triggers" if query.data == CallbackPrefix.TRIGGERS_RESET else "exceptions"
        
        await self._safe_answer_query(query, text="Выполняется сброс...")
        if item_type == "triggers": await self.db.reset_triggers()
        else: await self.db.reset_exceptions()
        
        return await self._show_item_management_menu(update, context, item_type)

        ban = rules_map.get(('BAN', None))
        
        rules_lines = [
            f"• Мьют на сутки \\- *{mute_24}* пред\\." if mute_24 else "• Мьют на сутки \\- *отключен*",
            f"• Мьют на неделю \\- *{mute_168}* пред\\." if mute_168 else "• Мьют на неделю \\- *отключен*",
            f"• Бан \\- *{ban}* пред\\." if ban else "• Бан \\- *отключен*"
        ]
        text = "🔨 *Настройка правил санкций*\n\n*Текущие правила:*\n" + "\n".join(rules_lines)
        
        kb = [
            [InlineKeyboardButton("Мьют на сутки", callback_data=f"{CallbackPrefix.SANCTION_SET}MUTE_24")],
            [InlineKeyboardButton("Мьют на неделю", callback_data=f"{CallbackPrefix.SANCTION_SET}MUTE_168")],
            [InlineKeyboardButton("Бан", callback_data=f"{CallbackPrefix.SANCTION_SET}BAN_None")],
            [InlineKeyboardButton("⬅️ Назад", callback_data=CallbackPrefix.BACK_TO_MODERATION_SETTINGS)]
        ]
        return text, InlineKeyboardMarkup(kb)

    async def show_sanctions_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Отображает меню управления санкциями."""
        query = update.callback_query
        await self._safe_answer_query(query)
        text, reply_markup = await self._get_sanctions_menu_content()
        await self._safe_edit_message(query, text, reply_markup=reply_markup, parse_mode=ParseMode.MARKDOWN_V2)
        return SANCTIONS_MENU

    async def sanction_request_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает количество предупреждений для установки правила санкции."""
        query = update.callback_query
        await self._safe_answer_query(query)
        
        data = query.data.replace(CallbackPrefix.SANCTION_SET, "")
        action, dur_str = data.split("_")
        duration = int(dur_str) if dur_str != "None" else None
        
        context.user_data["current_sanction"] = {"action": action, "duration_hours": duration}
        context.user_data["original_update"] = update
        
        txt = ""
        if action == 'BAN':
            txt = "бана"
        elif duration == 24:
            txt = "мьюта на сутки"
        elif duration == 168:
            txt = "мьюта на неделю"
        
        await self._safe_edit_message(
            query,
            f"Введите количество предупреждений для *{txt}*\\.\nЧисло `0` удалит правило\\.",
            reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("⬅️ Отмена", callback_data=CallbackPrefix.BACK_TO_SANCTIONS_MENU)]]),
            parse_mode=ParseMode.MARKDOWN_V2
        )
        return AWAIT_SANCTION_VALUE

    async def sanction_set_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Устанавливает новое значение для правила санкции."""
        msg = update.message
        try:
            warns = int(msg.text.strip())
            if warns < 0: raise ValueError
        except (ValueError, TypeError):
            await msg.reply_text("❌ Введите целое неотрицательное число\\.")
            return AWAIT_SANCTION_VALUE

        sanction = context.user_data.pop("current_sanction", None)
        if not sanction:
            await msg.reply_text("❌ Внутренняя ошибка\\.")
            return await self.show_moderate_settings_menu(update, context)

        rules = await self.db.get_sanction_rules()
        rules = [r for r in rules if not (r["action"] == sanction["action"] and r.get("duration_hours") == sanction.get("duration_hours"))]
        
        if warns > 0:
            rules.append({"warnings": warns, **sanction})
            
        await self.db.set_sanction_rules(rules)
        await context.bot.delete_message(msg.chat_id, msg.message_id)
        
        original_update = context.user_data.pop('original_update', None)
        if original_update and original_update.callback_query:
            await self._safe_answer_query(original_update.callback_query, "✅ Правила обновлены.")
            text, reply_markup = await self._get_sanctions_menu_content()
            await self._safe_edit_message(original_update.callback_query, text, reply_markup=reply_markup, parse_mode=ParseMode.MARKDOWN_V2)
            
        return SANCTIONS_MENU

    async def show_users_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Отображает меню пользователей с предупреждениями."""
        query = update.callback_query
        await self._safe_answer_query(query)
        
        users = await self.db.get_all_users_with_warnings()
        lines, kb = ["👥 *Пользователи с предупреждениями*\n"], []
        
        if not users:
            lines.append("Чисто — нарушителей нет\\.")
        else:
            for rec in users:
                uid, count = rec["user_id"], rec["count"]
                try:
                    tg_user = await context.bot.get_chat(uid)
                    uname = escape_markdown_v2(f"@{tg_user.username}" if tg_user.username else tg_user.full_name)
                    btn_uname = escape_markdown_v2(tg_user.username or tg_user.first_name)
                except Exception:
                    uname = f"`{uid}`"
                    btn_uname = f"ID: {uid}"
                lines.append(f"• {uname} — *{count}* пред\\.")
                kb.append([InlineKeyboardButton(f"❌ Сбросить для {btn_uname}", callback_data=f"{CallbackPrefix.RESET_WARNS}{uid}")])
                
        kb.append([InlineKeyboardButton("⬅️ Назад", callback_data=CallbackPrefix.BACK_TO_MODERATION_SETTINGS)])
        
        await self._safe_edit_message(query, "\n".join(lines), reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.MARKDOWN_V2)
        return USERS_MENU

    async def reset_user_warnings(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Сбрасывает предупреждения для конкретного пользователя."""
        query = update.callback_query
        await self._safe_answer_query(query)
        await self._disable_keyboard(query)
        
        uid = int(query.data.replace(CallbackPrefix.RESET_WARNS, ""))
        await self.db.reset_warnings(uid)
        await query.answer("Предупреждения пользователя сброшены.")
        
        return await self.show_users_menu(update, context)

"""
Обработчики инлайн-кнопок для управления системными промптами.

Version 1.9.0:
- FEATURE: В меню редактирования "Промпта ИИ-Фасилитатора" добавлена
           подсказка о доступности переменной `{{group_rules}}`.

Version 1.8.0:
- FIX: Заменен вызов `context.user_data.clear()` на безопасную функцию
       `clear_conversation_user_data` для предотвращения сброса состояния
       диалога и исправления бага с неработающими диалогами.
"""
from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode, MessageLimit
from telegram.ext import ContextTypes, ConversationHandler

from ...config import settings
from ...utils.formatters import safe_markdown_preview, escape_markdown_v2
from ...utils.context_helpers import clear_conversation_user_data
from ..states import (FACILITATOR_PROMPT_MENU, MODERATION_PROMPT_MENU, MODERATION_TEMPLATE_MENU,
                      WELCOME_PROMPT_MENU, AWAIT_PROMPT_INPUT)
from .callback_patterns import CallbackPrefix

class PromptCallbackHandler:
    """Обработчики для меню управления текстами промптов."""

    async def _get_prompt_data(self, text_type: str) -> tuple:
        if text_type == "facilitator_prompt":
            content = await self.db.get_prompt()
            title = "📜 *Промпт ИИ\\-Фасилитатора*"
            back_cb = CallbackPrefix.BACK_TO_FACILITATOR_SETTINGS
            ret_state = FACILITATOR_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_FACILITATOR_PROMPT
            reset_cb = CallbackPrefix.RESET_FACILITATOR_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "moderation_prompt":
            content = await self.db.get_moderation_prompt()
            title = "🔍 *Промпт Судьи*"
            back_cb = CallbackPrefix.BACK_TO_MODERATION_SETTINGS
            ret_state = MODERATION_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_MODERATION_PROMPT
            reset_cb = CallbackPrefix.RESET_MODERATION_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "moderation_template":
            content = await self.db.get_moderation_task_template()
            title = "📝 *Промпт Сообщения о Санкции*"
            back_cb = CallbackPrefix.BACK_TO_MODERATION_SETTINGS
            ret_state = MODERATION_TEMPLATE_MENU
            edit_cb = CallbackPrefix.EDIT_MODERATION_TEMPLATE
            reset_cb = CallbackPrefix.RESET_MODERATION_TEMPLATE
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "welcome_prompt":
            content = await self.db.get_welcome_prompt()
            title = "👋 *Промпт Приветствия*"
            back_cb = CallbackPrefix.BACK_TO_WELCOMES_MENU
            ret_state = WELCOME_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_WELCOME_PROMPT
            reset_cb = CallbackPrefix.RESET_WELCOME_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        return None, None, None, None, None, None

    async def _show_text_management_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE, text_type: str) -> int:
        query = update.callback_query
        if query:
            await self._safe_answer_query(query)
        
        text_content, title, back_cb, ret_state, edit_cb, reset_cb = await self._get_prompt_data(text_type)
        if text_content is None:
             return ConversationHandler.END

        kb = [[InlineKeyboardButton("✏️ Изменить", callback_data=edit_cb)]]
        
        preview = safe_markdown_preview(text_content, settings.PROMPT_PREVIEW_LENGTH)
        if len(text_content) > settings.PROMPT_PREVIEW_LENGTH:
            kb.append([InlineKeyboardButton("⬇️ Показать полностью", callback_data=f"{CallbackPrefix.SEND_FULL_PROMPT}{text_type}")])

        kb.extend([
            [InlineKeyboardButton("🔄 Сбросить", callback_data=reset_cb),
             InlineKeyboardButton("⬅️ Назад", callback_data=back_cb)]
        ])
        
        text = f"{title}:\n\n{preview}"
        
        await self._safe_edit_message(query, text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.MARKDOWN_V2)
        return ret_state

    async def send_full_prompt(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        prompt_type = query.data.replace(CallbackPrefix.SEND_FULL_PROMPT, "")
        
        text_content, title, _, ret_state, _, _ = await self._get_prompt_data(prompt_type)
        
        if not text_content:
            await query.answer("❌ Не удалось найти текст промпта.", show_alert=True)
            return ret_state

        await query.answer()

        await context.bot.send_message(
            chat_id=query.message.chat_id,
            text=f"Полный текст для: {title}",
            parse_mode=ParseMode.MARKDOWN_V2
        )

        for i in range(0, len(text_content), MessageLimit.MAX_TEXT_LENGTH):
            chunk = text_content[i:i + MessageLimit.MAX_TEXT_LENGTH]
            await context.bot.send_message(
                chat_id=query.message.chat_id,
                text=chunk
            )
        
        return ret_state

    async def show_facilitator_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "facilitator_prompt")

    async def show_moderation_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "moderation_prompt")
        
    async def show_moderation_template_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "moderation_template")

    async def show_welcome_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "welcome_prompt")

    async def request_prompt_input(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """
        Запрашивает у пользователя новый текст для промпта, не отображая текущий.
        Это предотвращает ошибки, связанные с превышением лимита длины сообщения.
        """
        query = update.callback_query
        await self._safe_answer_query(query)
        
        clear_conversation_user_data(context)

        cb_data = query.data
        prompt_type, prompt_text = "", ""
        
        if cb_data == CallbackPrefix.EDIT_FACILITATOR_PROMPT:
            prompt_type = "facilitator_prompt"
            prompt_text = (
                "Отправьте новый текст для *Промпта ИИ\\-Фасилитатора \\(Ядра Личности\\)*\\.\n\n"
                "*Доступные переменные:* `{{group_rules}}`\n\n"
                "Для отмены введите /cancel\\."
            )
        elif cb_data == CallbackPrefix.EDIT_MODERATION_PROMPT:
            prompt_type = "moderation_prompt"
            prompt_text = "Отправьте новый текст для *Промпта Судьи*\\.\n\n*Обязательная переменная:* `{{text}}`\n\nДля отмены введите /cancel\\."
        elif cb_data == CallbackPrefix.EDIT_MODERATION_TEMPLATE:
            prompt_type = "moderation_template"
            prompt_text = "Отправьте новый текст для *Промпта Сообщения о Санкции*\\.\n\n*Доступные переменные:* `{{user_mention}}`, `{{violator_message}}`, `{{warnings_count}}`, `{{rules_list}}`, `{{chat_history}}`, `{{group_rules}}`\n\nДля отмены введите /cancel\\."
        elif cb_data == CallbackPrefix.EDIT_WELCOME_PROMPT:
            prompt_type = "welcome_prompt"
            prompt_text = "Отправьте новый текст для *Промпта Приветствия*\\.\n\n*Обязательная переменная:* `{{user_mention}}`\n\nДля отмены введите /cancel\\."
        else:
            return ConversationHandler.END
            
        context.user_data["prompt_type"] = prompt_type
        context.user_data["original_update"] = update
        
        await self._safe_edit_message(query, prompt_text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        
        return AWAIT_PROMPT_INPUT

    async def reset_prompt_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        await self._safe_answer_query(query)
        await self._disable_keyboard(query)
        
        cb_data = query.data
        if cb_data == CallbackPrefix.RESET_FACILITATOR_PROMPT:
            await self.db.reset_prompt()
            await query.answer("✅ Промпт Фасилитатора сброшен.")
            return await self.show_facilitator_prompt_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_MODERATION_PROMPT:
            await self.db.reset_moderation_prompt()
            await query.answer("✅ Промпт Судьи сброшен.")
            return await self.show_moderation_prompt_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_MODERATION_TEMPLATE:
            await self.db.reset_moderation_task_template()
            await query.answer("✅ Промпт Сообщения о Санкции сброшен.")
            return await self.show_moderation_template_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_WELCOME_PROMPT:
            await self.db.reset_welcome_prompt()
            await query.answer("✅ Промпт Приветствия сброшен.")
            return await self.show_welcome_prompt_menu(update, context)
            
        return ConversationHandler.END

            FACILITATOR_SETTINGS: [

                CallbackQueryHandler(callback_handlers.show_generation_settings_menu, pattern=f"^{CallbackPrefix.GENERATION_SETTINGS}$"),
                CallbackQueryHandler(callback_handlers.show_context_settings_menu, pattern=f"^{CallbackPrefix.CONTEXT_SETTINGS}$"),
                CallbackQueryHandler(callback_handlers.toggle_debug_mode, pattern=f"^{CallbackPrefix.TOGGLE_DEBUG_MODE}$"),
            ],
            FACILITATOR_PROMPT_MENU: [ *prompt_menu_handlers, CallbackQueryHandler(callback_handlers.show_facilitator_settings_menu, pattern=f"^{CallbackPrefix.BACK_TO_FACILITATOR_SETTINGS}$"), ],

"""
Обработчики команд и сообщений от обычных пользователей.

Version 5.0.0:
- REFACTOR: Адаптация к композитной архитектуре промптов.
            - `_respond_to_message` и `greet_new_members` теперь передают
              `task_instruction` вместо `system_prompt` в `GeminiService`.
            - Реализована логика получения описания чата для подстановки
              переменной `{{group_rules}}`.

Version 4.9.1:
- REFACTOR: Унифицирована логика форматирования. Теперь ParseMode.MARKDOWN
            используется как для обычного, так и для отладочного режима,
            чтобы обеспечить консистентное отображение ответов.
"""
import asyncio
import logging
import uuid
import json
from telegram import Update, ReplyParameters
from telegram.ext import ContextTypes
from telegram.constants import ChatType, ParseMode, ChatAction, MessageLimit
from telegram.error import BadRequest, TelegramError

from ..database import Database
from ..services.gemini import GeminiService

from ..utils.decorators import log_action
from ..utils.formatters import format_debug_info_text, escape_markdown_v2

logger = logging.getLogger(__name__)

class UserHandlers:
    """Класс, инкапсулирующий обработчики пользовательских команд."""

    def __init__(self, db: Database, gemini_service: GeminiService, moderation_service: ModerationService):
        self.db = db
        self.gemini = gemini_service
        self.moderation = moderation_service

    @log_action("message")
    async def handle_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        message = update.message
        user = update.effective_user
        chat = update.effective_chat
        text = message.text
        trace_id = str(uuid.uuid4())
        context.user_data['trace_id'] = trace_id

        if not all([message, user, chat, text]):
            return

        bot_username = context.bot.username.lower()
        is_mention = f"@{bot_username}" in text.lower()
        is_reply_to_bot = message.reply_to_message and message.reply_to_message.from_user.id == context.bot.id
        
        should_respond = False
        is_admin_private_chat = chat.type == ChatType.PRIVATE and user.id in ADMIN_IDS_SET

        if is_admin_private_chat:
            is_debug_mode = await self.db.is_debug_mode_enabled()
            if is_debug_mode:
                if is_mention or is_reply_to_bot:
                    logger.info("Причина: Админ в ЛС (Отладка ВКЛ), прямое обращение. Бот отвечает.", extra={'trace_id': trace_id})
                    should_respond = True
                else:
                    logger.info(f"Причина: Админ в ЛС (Отладка ВКЛ), заметка тестера. Логгирую, не отвечаю. Текст: «{text}»", extra={'trace_id': trace_id})
                    await self.db.add_message(chat.id, user.full_name, text, is_bot=False)
                    should_respond = False
            else:
                logger.info("Причина: Админ в ЛС (Отладка ВЫКЛ), штатный режим. Бот отвечает.", extra={'trace_id': trace_id})
                should_respond = True
        
        elif chat.type in [ChatType.GROUP, ChatType.SUPERGROUP]:
            if is_mention or is_reply_to_bot:
                logger.info("Причина: Группа + упоминание/ответ. Бот отвечает.", extra={'trace_id': trace_id})
                should_respond = True
            else:
                logger.info("Причина: Группа, нет упоминания. Передаю модератору.", extra={'trace_id': trace_id})
                is_violation = await self.moderation.check_message(update, context, trace_id)
                if is_violation:
                    await self.moderation.handle_violation(update, context, text, trace_id)
                else:
                    await self.db.add_message(chat.id, user.full_name, text, is_bot=False)
        
        elif chat.type == ChatType.PRIVATE:
            logger.info("Причина: Обычный пользователь в ЛС. Отвечаю заглушкой.", extra={'trace_id': trace_id})
            await self.db.add_message(chat.id, user.full_name, text, is_bot=False)
            await update.message.reply_text(PRIVATE_CHAT_REPLY)
            should_respond = False

        if should_respond:
            await self._respond_to_message(update, context, trace_id)

    @log_action("start")
    async def start_command(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        user = update.effective_user
        chat = update.effective_chat
        if not user or not chat: return

        if chat.type == ChatType.PRIVATE:
            if user.id in ADMIN_IDS_SET:
                await update.message.reply_text(ADMIN_START_MESSAGE)
            else:
                await update.message.reply_text(PRIVATE_CHAT_REPLY)
        elif chat.type in [ChatType.GROUP, ChatType.SUPERGROUP]:
            await update.message.reply_text(GROUP_START_MESSAGE.format(bot_username=context.bot.username))

    @log_action("help")
    async def help_command(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        text = (
            "Я — бот, помогающий практиковать ненасильственное общение\\.\n\n"
            "**В групповых чатах:**\n"
            f"• Чтобы задать мне вопрос, упомяните меня \\(`@{escape_markdown_v2(context.bot.username)}`\\) или ответьте на моё сообщение\\.\n"
            "• Я также слежу за общением и могу реагировать на некоторые сообщения\\.\n\n"
            "**В личных сообщениях \\(только для администраторов\\):**\n"
            "/settings \\- Настройки личности ИИ\\-бота\\.\n"
            "/moderate \\- Настройки модерации\\.\n"
            "/stats \\- Статистика\\.\n"
            "/warnings \\- Управление предупреждениями\\.\n"
            "/cancel \\- Отмена диалога\\."
        )
        await update.message.reply_text(text, parse_mode=ParseMode.MARKDOWN_V2)

    @log_action("greet_new_members")
    async def greet_new_members(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        if not await self.db.is_welcome_enabled(): return
        if not update.message or not update.message.new_chat_members: return
        
        trace_id = str(uuid.uuid4())
        chat = update.effective_chat
        
        try:
            chat_full = await context.bot.get_chat(chat.id)
            chat_description = chat_full.description or "Правила группы не указаны."
        except Exception as e:
            logger.warning("Не удалось получить описание чата %d: %s", chat.id, e, extra={'trace_id': trace_id})
            chat_description = "Правила группы не указаны."

        for member in update.message.new_chat_members:
            if member.is_bot: continue
            logger.info("Генерация приветствия для %s (%d)", member.full_name, member.id, extra={'trace_id': trace_id})
            try:
                task_instruction = await self.db.get_welcome_prompt()
                if not task_instruction:
                    logger.warning("Промпт приветствий пуст.", extra={'trace_id': trace_id})
                    continue

                # Подставляем переменные в инструкцию
                formatted_instruction = task_instruction.format(user_mention=member.mention_html()).replace("@имя_бота", f"@{context.bot.username}")
                
                model_id = await self.db.get_welcome_model()
                
                # Для приветствия "вопрос" пуст, вся задача описана в инструкции
                response_text, _ = await self.gemini.generate_response(
                    question="",
                    task_instruction=formatted_instruction,
                    trace_id=trace_id,
                    model_override=model_id,
                    chat_description=chat_description
                )
                if response_text:
                    await context.bot.send_message(chat_id=chat.id, text=response_text, parse_mode=ParseMode.HTML)
            except Exception as e:
                logger.error("Ошибка при генерации приветствия: %s", e, extra={'trace_id': trace_id})

    async def _keep_typing_action(self, context: ContextTypes.DEFAULT_TYPE, chat_id: int, trace_id: str) -> None:
        while True:
            try:
                await context.bot.send_chat_action(chat_id=chat_id, action=ChatAction.TYPING)
                await asyncio.sleep(4)
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.warning("Ошибка в задаче 'typing': %s", e, extra={'trace_id': trace_id})
                break

    async def _send_long_message(self, context: ContextTypes.DEFAULT_TYPE, chat_id: int, text: str, trace_id: str, **kwargs):
        if not text:
            logger.warning("Попытка отправить пустое сообщение.", extra={'trace_id': trace_id})
            return
        
        for i in range(0, len(text), MessageLimit.MAX_TEXT_LENGTH):
            chunk = text[i:i + MessageLimit.MAX_TEXT_LENGTH]
            await context.bot.send_message(chat_id=chat_id, text=chunk, **kwargs)

    async def _respond_to_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE, trace_id: str) -> None:
        user = update.effective_user
        message = update.message
        chat_id = message.chat_id
        
        question = message.text
        typing_task = asyncio.create_task(self._keep_typing_action(context, chat_id, trace_id))
        
        try:
            # Получаем описание чата для переменной {{group_rules}}
            try:
                chat_full = await context.bot.get_chat(chat_id)
                chat_description = chat_full.description or "Правила группы не указаны."
            except Exception as e:
                logger.warning("Не удалось получить описание чата %d: %s", chat_id, e, extra={'trace_id': trace_id})
                chat_description = "Правила группы не указаны."

            context_config = await self.db.get_context_config()
            chat_history = await self.db.get_chat_history(chat_id, context_config["context_messages_limit"])
            user_mention = f"@{user.username}" if user.username else user.full_name
            is_debug_mode = await self.db.is_debug_mode_enabled()

            # Для обычного ответа инструкция пуста, т.к. всё описано в "Ядре личности"
            response_text, usage_metadata = await self.gemini.generate_response(
                question=question,
                task_instruction="",
                chat_history=chat_history,
                trace_id=trace_id,
                user_mention=user_mention,
                is_debug_mode=is_debug_mode,
                chat_description=chat_description
            )
            
            await self.db.add_message(chat_id, user.full_name, question, is_bot=False)
            if response_text:
                await self.db.add_message(chat_id, "bot", response_text, is_bot=True)
                await self.db.increment_ai_responses()
            
            parse_mode_for_response = ParseMode.MARKDOWN
            
            reply_params = ReplyParameters(message_id=message.message_id)
            try:
                await self._send_long_message(context, chat_id, response_text, trace_id, parse_mode=parse_mode_for_response, reply_parameters=reply_params)
            except BadRequest as e:
                if "Can't parse entities" in str(e):
                    logger.warning("Ошибка парсинга Markdown в ответе AI, отправляю как простой текст: %s", e, extra={'trace_id': trace_id})
                    await self._send_long_message(context, chat_id, response_text, trace_id, reply_parameters=reply_params)
                else:
                    logger.error("Ошибка BadRequest при отправке ответа AI: %s", e, extra={'trace_id': trace_id})
                    await message.reply_text("Произошла ошибка при отправке ответа.")

            if is_debug_mode and user.id in ADMIN_IDS_SET:
                # В `format_debug_info_text` больше не нужно передавать длины промптов,
                # так как эта информация теперь логируется в `ai_requests.log`
                debug_info = format_debug_info_text(
                    usage_metadata=usage_metadata,
                    system_prompt_len=0, # Устарело, но оставим для совместимости
                    chat_history_len=len(json.dumps(chat_history)),
                    question_len=len(question)
                )
                await self._send_long_message(context, chat_id, debug_info, trace_id, parse_mode=ParseMode.MARKDOWN_V2)

        except Exception as e:
            logger.error("Критическая ошибка при генерации ответа: %s", e, exc_info=True, extra={'trace_id': trace_id})
            await message.reply_text("Произошла внутренняя ошибка.")
        finally:
            typing_task.cancel()

async def error_handler(update: object, context: ContextTypes.DEFAULT_TYPE) -> None:
    trace_id = context.user_data.get('trace_id', str(uuid.uuid4()))
    logger.error("Exception while handling an update:", exc_info=context.error, extra={'trace_id': trace_id})

    if not ADMIN_IDS: return

    tb_list = traceback.format_exception(None, context.error, context.error.__traceback__)
    tb_string = "".join(tb_list)

    update_summary = "N/A"
    if isinstance(update, Update):
        parts = []
        if update.effective_chat: parts.append(f"Chat: {update.effective_chat.title or 'Private'} ({update.effective_chat.id})")
        if update.effective_user: parts.append(f"User: {update.effective_user.full_name} ({update.effective_user.id})")
        if update.callback_query: parts.append(f"Callback: {update.callback_query.data}")
        elif update.effective_message: parts.append(f"Message: {update.effective_message.text[:70] if update.effective_message.text else '<No Text>'}")
        update_summary = " | ".join(parts)

    info_message = (
        f"<b>❗️ Исключение при обработке апдейта</b>\n\n"
        f"<b>Trace ID:</b> <code>{html.escape(trace_id)}</code>\n"
        f"<b>Сводка:</b>\n<pre>{html.escape(update_summary)}</pre>\n\n"
        f"<b>User Data:</b>\n<pre>{html.escape(str(context.user_data))}</pre>"
    )
    
    try:
        await context.bot.send_message(chat_id=ADMIN_IDS[0], text=info_message, parse_mode=ParseMode.HTML)
    except Exception as e:
        logger.error("Не удалось отправить первую часть сообщения об ошибке: %s", e, extra={'trace_id': trace_id})

    traceback_header = "\n\n<pre>------- TRACEBACK -------\n"
    chunk_size = 4096 - len(traceback_header) - len("</pre>")
    
    for i in range(0, len(tb_string), chunk_size):
        chunk = tb_string[i:i+chunk_size]
        message_chunk = f"{traceback_header if i == 0 else '<pre>'}{html.escape(chunk)}</pre>"
        try:
            await context.bot.send_message(chat_id=ADMIN_IDS[0], text=message_chunk, parse_mode=ParseMode.HTML)
        except Exception as e:
            logger.error("Не удалось отправить чанк traceback: %s", e, extra={'trace_id': trace_id})

async def main() -> None:
    """
    Главная асинхронная функция, которая настраивает и запускает бота.
    """
    setup_logging()
    
    db = Database()
    await db.initialize()
    await db.populate_defaults()

    gemini_service = GeminiService(
        model_names={"pro": settings.GEMINI_PRO_MODEL, "flash": settings.GEMINI_FLASH_MODEL, "lite": settings.GEMINI_LITE_MODEL},

"""
Сервис-обертка для взаимодействия с Google Gemini API.

Version 3.1.0:
- FIX: Устранена ошибка `ValueError: 'content' argument must not be empty`.
         Метод `generate_response` теперь корректно обрабатывает случаи, когда
         `question` пуст (например, при генерации приветствия), подставляя
         нейтральный текст для инициации ответа модели.

Version 3.0.0:
- REFACTOR: Внедрена композитная архитектура промптов.
            - Метод `generate_response` теперь принимает `task_instruction` вместо `system_prompt`.
            - Финальный системный промпт динамически собирается из "Ядра личности" (из БД)
              и "Инструкции к задаче", обеспечивая консистентность поведения.
            - Добавлена обработка переменной `{{group_rules}}` через новый
              аргумент `chat_description`.
- NOTE: Метод `get_moderation_verdict` намеренно не затронут и не использует
        "Ядро личности" для сохранения своей технической точности.
"""
import logging
import json
from typing import Dict, List, Optional, Tuple, Any

import google.generativeai as genai
from google.generativeai.types import GenerationConfig, content_types
from google.generativeai import types as genai_types

from ..config import settings
from ..database import Database

logger = logging.getLogger(__name__)
ai_logger = logging.getLogger('ai_requests')

class GeminiService:
    """
    Инкапсулирует логику работы с моделями Gemini.
    """

    def __init__(self, model_names: Dict[str, str], db: Database):
        self.db = db
        try:
            genai.configure(api_key=settings.GEMINI_API_KEY)
            logger.info("Ключ Gemini API успешно сконфигурирован.", extra={'trace_id': 'init'})
        except Exception as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА: Не удалось сконфигурировать Gemini API.", exc_info=e, extra={'trace_id': 'init'})
            raise

        self.model_names = model_names
        self.models: Dict[str, str] = {}
        for name, model_id in model_names.items():
            try:
                self.models[model_id] = model_id
                logger.info("Модель '%s' зарегистрирована для использования.", model_id, extra={'trace_id': 'init'})
            except Exception as e:
                logger.error("Ошибка при регистрации модели %s: %s", model_id, e, extra={'trace_id': 'init'})
        
        if not self.models:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА: Не удалось зарегистрировать ни одну AI-модель.", extra={'trace_id': 'init'})
            raise RuntimeError("Не удалось инициализировать AI-модели.")

        self.current_model_id = model_names.get("pro", next(iter(self.models.keys())))

    def switch_model(self, model_id: str) -> bool:
        if model_id in self.models:
            self.current_model_id = model_id
            logger.info("Активная модель переключена на: %s", model_id, extra={'trace_id': 'config'})
            return True
        logger.warning("Попытка переключиться на несуществующую модель: %s", model_id, extra={'trace_id': 'config'})
        return False

    def get_current_model(self) -> str:
        return self.current_model_id

    def get_available_models(self) -> List[str]:
        return list(self.models.keys())

    def _format_history_for_log(self, history: List[Dict]) -> str:
        """Форматирует историю чата в компактную строку для логов."""
        if not history:
            return "[]"
        return "\n".join([f"  [{item.get('role')}] {item.get('parts', [{}])[0].get('text', '')[:120]}..." for item in history])

    async def generate_response(
        self,
        question: str,
        task_instruction: str,
        trace_id: str,
        chat_history: List[Dict] = None,
        user_mention: Optional[str] = None,
        model_override: Optional[str] = None,
        is_debug_mode: bool = False,
        chat_description: Optional[str] = None
    ) -> Tuple[str, Optional[Dict[str, Any]]]:
        model_id_to_use = model_override if model_override and model_override in self.models else self.current_model_id
        
        try:
            # 1. Загружаем "Ядро личности"
            core_prompt = await self.db.get_prompt()
            
            # 2. Подставляем в "Ядро" правила группы, если они переданы
            if chat_description:
                core_prompt = core_prompt.replace("{{group_rules}}", chat_description)

            # 3. Собираем финальный системный промпт
            final_system_prompt = core_prompt
            if task_instruction and task_instruction.strip():
                final_system_prompt = f"{core_prompt}\n\n---\n\n{task_instruction}"

            model = genai.GenerativeModel(
                model_name=model_id_to_use,
                system_instruction=final_system_prompt
            )
            
            gen_config_dict = await self.db.get_generation_config()
            
            thinking_enabled_log_status = "OFF"
            if is_debug_mode:
                try:
                    gen_config_dict['thinking_config'] = genai_types.ThinkingConfig(include_thoughts=True)
                    thinking_enabled_log_status = "ON"
                except AttributeError:
                    logger.warning("Не удалось включить 'мысли': `ThinkingConfig` не найден в текущей версии SDK. Обновите google-generativeai.", extra={'trace_id': trace_id})
                    thinking_enabled_log_status = "UNSUPPORTED"

            generation_config = GenerationConfig(**gen_config_dict)
            
            history = chat_history or []
            chat_session = model.start_chat(history=history)
            
            # FIX: Обеспечиваем, чтобы `user_question_text` никогда не был пустым.
            if question:
                user_question_text = f"Сообщение от пользователя {user_mention}: {question}" if user_mention else question
            else:
                # Для задач без прямого вопроса (приветствие), инициируем ответ нейтральной фразой.
                # Задача генерации уже полностью описана в `final_system_prompt`.
                user_question_text = "Продолжай согласно инструкции."
            
            logger.info("Отправка запроса к модели %s (Thinking: %s)...", model_id_to_use, thinking_enabled_log_status, extra={'trace_id': trace_id})
            
            response = await chat_session.send_message_async(
                content=user_question_text,
                generation_config=generation_config
            )
            
            usage_metadata_obj = getattr(response, 'usage_metadata', None)
            full_usage_metadata = {}
            if usage_metadata_obj:
                all_fields = [
                    "prompt_token_count", "cached_content_token_count", 
                    "candidates_token_count", "total_token_count",
                    "tool_use_prompt_token_count", "thoughts_token_count",
                ]
                for field in all_fields:
                    full_usage_metadata[field] = getattr(usage_metadata_obj, field, None)
                logger.info("Полные метаданные об использовании токенов извлечены.", extra={'trace_id': trace_id})
            else:
                logger.warning("Атрибут 'usage_metadata' не найден в ответе от Gemini API.", extra={'trace_id': trace_id})

            if is_debug_mode:
                history_str = json.dumps(history, ensure_ascii=False)
                
                thoughts_log = ""
                try:
                    if response.candidates:
                        for candidate in response.candidates:
                            if candidate.content and candidate.content.parts:
                                for part in candidate.content.parts:
                                    if hasattr(part, 'thought') and part.thought:
                                        thoughts_log += f"\n{part.thought}"
                except Exception as e:
                    thoughts_log = f" (ошибка извлечения: {e})"

                log_message = (
                    f"--- AI REQUEST (DEBUG) ---\n"
                    f"MODEL: {model_id_to_use}\n"
                    f"--- FINAL SYSTEM PROMPT ({len(final_system_prompt)} chars) ---\n{final_system_prompt}\n"
                    f"--- CHAT HISTORY ({len(history_str)} chars) ---\n{self._format_history_for_log(history)}\n"
                    f"--- QUESTION ({len(user_question_text)} chars) ---\n{user_question_text}\n"
                    f"--- THOUGHTS ---{thoughts_log if thoughts_log else ' (отсутствуют или не поддерживаются SDK)'}\n"
                    f"--- USAGE ---\n{json.dumps(full_usage_metadata, default=str, indent=2)}\n"
                    f"--- END REQUEST ---"
                )
            else:
                log_message = (
                    f"--- AI REQUEST ---\n"
                    f"MODEL: {model_id_to_use}\n"
                    f"FINAL_SYSTEM_PROMPT_LEN: {len(final_system_prompt)}\n"
                    f"CHAT_HISTORY_LEN: {len(json.dumps(history))}\n"
                    f"QUESTION: {user_question_text[:150]}...\n"
                    f"USAGE: total_tokens={full_usage_metadata.get('total_token_count', 'N/A')}\n"
                    f"--- END REQUEST ---"
                )
            ai_logger.info(log_message, extra={'trace_id': trace_id})

            return response.text, full_usage_metadata

        except Exception as e:
            logger.error("Ошибка при генерации ответа от Gemini: %s", e, exc_info=True, extra={'trace_id': trace_id})
            return f"Произошла ошибка при обращении к AI. Попробуйте позже. ({type(e).__name__})", None

    async def get_moderation_verdict(self, text: str, prompt_template: str, model_id: str, trace_id: str) -> bool:
        model_id_to_use = self.models.get(model_id)
        if not model_id_to_use:
            logger.error("Модель модерации %s не найдена.", model_id, extra={'trace_id': trace_id})
            return False
        
        model = genai.GenerativeModel(model_id_to_use)
        prompt = prompt_template.format(text=text)
        
        try:
            moderation_config = GenerationConfig(temperature=0.0, top_p=1.0, top_k=1, max_output_tokens=10)
            response = await model.generate_content_async([prompt], generation_config=moderation_config)
            verdict = response.text.strip().upper()
            logger.debug("Ответ модератора для текста '%s...': '%s'", text[:50], verdict, extra={'trace_id': trace_id})
            return verdict == "ДА"
        except Exception as e:
            logger.error("Ошибка при получении вердикта модерации: %s", e, extra={'trace_id': trace_id})
            return False

"""
Сервис, отвечающий за модерацию сообщений в чате.

Version 2.0.0:
- REFACTOR: Адаптация к композитной архитектуре промптов.

            - Реализована подстановка переменной `{{group_rules}}`.
            - История чата теперь передается как часть инструкции, а не отдельным
              аргументом, что упрощает вызов сервиса.

Version 1.8.1:
- REFACTOR: Адаптация к новому формату `get_chat_history`, возвращающему
            `List[Dict]`. Добавлена внутренняя функция для преобразования
            структурированной истории в строку для шаблона.
"""
import logging
import re
from datetime import datetime, timedelta
from typing import Dict, List, Any
from telegram import Update, ReplyParameters
from telegram.ext import ContextTypes
from telegram.error import TelegramError, BadRequest
from telegram.constants import ParseMode

from ..database import Database
from .gemini import GeminiService
from ..config import settings

logger = logging.getLogger(__name__)

class ModerationService:
    """
    Инкапсулирует логику проверки сообщений и применения санкций.
    """

    def __init__(self, db: Database, gemini: GeminiService):
        self.db = db
        self.gemini = gemini

    async def check_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE, trace_id: str) -> bool:
        if not await self.db.is_moderation_enabled():
            return False

        user = update.effective_user
        chat = update.effective_chat
        text = update.message.text

        if not all([user, chat, text]):
            return False
        
        text_lower = text.lower()

        try:
            chat_admins = await context.bot.get_chat_administrators(chat.id)
            admin_ids = {admin.user.id for admin in chat_admins}
            if user.id in admin_ids:
                return False
        except TelegramError as e:
            logger.warning("Не удалось получить список админов чата %d: %s", chat.id, e, extra={'trace_id': trace_id})

        triggers = await self.db.get_all_triggers_flat()
        exceptions = await self.db.get_all_exceptions_flat()

        if not any(re.search(r'\b' + re.escape(trigger) + r'\b', text_lower) for trigger in triggers):
            return False

        if any(exc in text_lower for exc in exceptions):
            logger.debug("Найден триггер и исключение. Проверка AI отменена.", extra={'trace_id': trace_id})
            return False

        logger.info("Сообщение передается AI-судье.", extra={'trace_id': trace_id})
        prompt_template = await self.db.get_moderation_prompt()
        model_id = await self.db.get_moderation_model()
        
        return await self.gemini.get_moderation_verdict(text, prompt_template, model_id, trace_id)

    def _format_rules_list(self, rules: List[Dict]) -> str:
        if not rules: return "Правила санкций не настроены."
        lines = []
        for rule in sorted(rules, key=lambda x: x['warnings']):
            action_desc = "бан" if rule['action'] == 'BAN' else f"мьют на {rule['duration_hours']} ч."
            lines.append(f"- {rule['warnings']} предупреждений: {action_desc}")
        return "\n".join(lines)

    def _format_history_for_template(self, history: List[Dict[str, Any]]) -> str:
        """Преобразует структурированную историю в простую строку."""
        if not history:
            return "Контекст диалога пуст."
        
        lines = []
        for item in history:
            role = item.get("role", "user")
            text = item.get("parts", [{}])[0].get("text", "")
            if role == "user":
                lines.append(f"user: {text}")
            else:
                lines.append(f"bot: {text}")
        return "\n".join(lines)

    async def handle_violation(self, update: Update, context: ContextTypes.DEFAULT_TYPE, text: str, trace_id: str) -> None:
        user = update.effective_user
        chat = update.effective_chat
        message = update.message
        if not all([user, chat, message]): return

        await self.db.increment_moderation_actions()
        new_warnings_count = await self.db.adjust_warnings(user.id, 1)
        logger.info("Пользователю %s выдано предупреждение. Всего: %d", user.id, new_warnings_count, extra={'trace_id': trace_id})
        
        sanction_rules = await self.db.get_sanction_rules()
        applied_sanction = next((r for r in sorted(sanction_rules, key=lambda x: x['warnings'], reverse=True) if new_warnings_count >= r['warnings']), None)

        if applied_sanction:
            action, duration = applied_sanction['action'], applied_sanction.get('duration_hours')
            try:
                if action == 'MUTE' and duration:
                    await context.bot.restrict_chat_member(chat.id, user.id, permissions={'can_send_messages': False}, until_date=datetime.now() + timedelta(hours=duration))
                    logger.info("Пользователь %s замьючен в чате %s на %d часов.", user.id, chat.id, duration, extra={'trace_id': trace_id})
                elif action == 'BAN':
                    await context.bot.ban_chat_member(chat.id, user.id)
                    logger.info("Пользователь %s забанен в чате %s.", user.id, chat.id, extra={'trace_id': trace_id})
            except Exception as e:
                logger.error("Не удалось применить санкцию '%s' для %s: %s", action, user.id, e, extra={'trace_id': trace_id})

        try:
            # Получаем описание чата для переменной {{group_rules}}
            try:
                chat_full = await context.bot.get_chat(chat.id)
                chat_description = chat_full.description or "Правила группы не указаны."
            except Exception as e:
                logger.warning("Не удалось получить описание чата %d: %s", chat.id, e, extra={'trace_id': trace_id})
                chat_description = "Правила группы не указаны."

            task_instruction_template = await self.db.get_moderation_task_template()
            context_config = await self.db.get_context_config()
            
            history_list = await self.db.get_chat_history(chat.id, context_config["context_judge_limit"])
            history_str = self._format_history_for_template(history_list)
            
            # Собираем инструкцию, подставляя все переменные
            formatted_instruction = task_instruction_template.format(
                user_mention=user.mention_markdown_v2(), violator_message=text,
                warnings_count=new_warnings_count, rules_list=self._format_rules_list(sanction_rules),
                chat_history=history_str, group_rules=chat_description
            )
            
            # "Вопрос" пуст, так как вся задача полностью описана в инструкции
            ai_response, _ = await self.gemini.generate_response(
                question="",
                task_instruction=formatted_instruction,
                trace_id=trace_id
            )
        except Exception as e:
            logger.error("Ошибка при генерации ответа AI-фасилитатором: %s", e, exc_info=True, extra={'trace_id': trace_id})
            ai_response = f"⚠️ {user.mention_markdown_v2()}, ваше сообщение было расценено как 'ты-сообщение'. У вас {new_warnings_count} предупреждение(й)."

        if ai_response:
            try:
                await message.reply_text(text=ai_response, parse_mode=ParseMode.MARKDOWN)
            except BadRequest as e:
                logger.error("Ошибка BadRequest при отправке сообщения о нарушении: %s", e, extra={'trace_id': trace_id})

"""
Вспомогательные функции для работы с контекстом ConversationHandler.

Этот модуль предоставляет безопасные способы управления данными в `context.user_data`,
чтобы избежать случайного удаления внутренних ключей, используемых библиотекой
python-telegram-bot для отслеживания состояния диалога.

Зачем это нужно:
Прямой вызов `context.user_data.clear()` уничтожает ВСЕ данные, включая
служебные ключи, которые `ConversationHandler` использует для ведения
диалога. Это приводит к "амнезии" обработчика и непредсказуемому поведению.
Эта функция решает проблему, удаляя только те ключи, которые были явно
добавлены логикой нашего приложения.
"""
from telegram.ext import ContextTypes
import logging

logger = logging.getLogger(__name__)

def clear_conversation_user_data(context: ContextTypes.DEFAULT_TYPE) -> None:
    """
    Безопасно очищает только те ключи из user_data, которые были установлены
    в ходе диалогов настроек, не затрагивая внутренние ключи PTB.

    Args:
        context (ContextTypes.DEFAULT_TYPE): Контекст бота, содержащий user_data.
    """
    # Список ключей, которые устанавливаются логикой нашего приложения в user_data
    # и должны быть очищены по завершении или отмене диалога.
    app_keys_to_clear = [
        "prompt_type",
        "prompt_buffer",
        "original_update",
        "prompt_commit_task",
        "context_param_key",
        "gen_param_key",
        "current_sanction",
        "trace_id",
    ]
    
    logger.debug("Очистка пользовательских ключей из user_data: %s", app_keys_to_clear)
    
    for key in app_keys_to_clear:
        if key in context.user_data:
            try:
                del context.user_data[key]
            except KeyError:
                # Это может произойти в редких случаях (например, при гонке состояний),
                # но это безопасно проигнорировать.
                pass

"""

# Логика парсинга и обработки ошибок теперь централизована в config.py.
ADMIN_IDS_SET = set(ADMIN_IDS)
if not ADMIN_IDS_SET:
    logger.warning("Список ADMIN_IDS пуст или не задан в .env. "
                   "Административные команды будут недоступны.")


def admin_only(func: Callable) -> Callable:
    """
    Декоратор, который проверяет, является ли пользователь администратором.
    Разрешает выполнение функции только для пользователей, чьи ID есть в ADMIN_IDS_SET.
    """
    @functools.wraps(func)
    async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
        """Внутренняя обертка, выполняющая проверку."""
        if not update.effective_user:
            logger.warning("Не удалось определить пользователя для проверки прав доступа.")
            return None
        
        user_id = update.effective_user.id
        
        # --- ИСПРАВЛЕНИЕ: Используем ADMIN_IDS_SET для проверки ---
        if user_id not in ADMIN_IDS_SET:
            logger.warning("Пользователь %d попытался выполнить команду только для администраторов: %s",
                         user_id, func.__name__)
            # Проверяем, есть ли у сообщения атрибут message, чтобы ответить
            if update.message:
                await update.message.reply_text("⛔ У вас нет прав для выполнения этой команды.")
            return None
            
        return await func(self, update, context, *args, **kwargs)
        
    return wrapper


def log_action(action_name: str) -> Callable:
    """
    Декоратор для логирования действий пользователя.
    
    Args:
        action_name (str): Название действия для отображения в логе.
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
            """Внутренняя обертка, выполняющая логирование."""
            user = update.effective_user
            chat = update.effective_chat
            
            if user and chat:
                log_message = (
                    f"Действие '{action_name}' вызвано пользователем {user.full_name} "
                    f"(ID: {user.id}, Username: @{user.username}) "
                    f"в чате '{getattr(chat, 'title', 'личный чат')}' (ID: {chat.id}, Type: {chat.type})."
                )
                logger.info(log_message)
            else:
                logger.info("Действие '%s' вызвано без определения пользователя/чата.", action_name)

            return await func(self, update, context, *args, **kwargs)
            
        return wrapper
    return decorator


def typing_action(func: Callable) -> Callable:
    """
    Декоратор, который отправляет действие 'typing...' в чат.
    Это дает пользователю понять, что бот обрабатывает запрос.
    """
    @functools.wraps(func)
    async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
        """Внутренняя обертка, отправляющая действие."""
        if update.effective_chat:
            # Отправляем действие "печатает..." перед выполнением основной логики
            await context.bot.send_chat_action(
                chat_id=update.effective_chat.id,
                action=ChatAction.TYPING
            )
        # Выполняем оригинальную функцию
        return await func(self, update, context, *args, **kwargs)
        
    return wrapper

"""
Модуль с утилитарными функциями для форматирования текста.

Version 1.9.0:
- REFACTOR: `format_debug_info_text` теперь скрывает поля токенов с нулевыми
            значениями в отладочном выводе в чате для улучшения читаемости.
Version 1.8.0:
- REFACTOR: `format_debug_info_text` полностью переписана.
            - Теперь отображает все ключевые поля токенов из документации API.
            - Вычисляет и показывает "Неучтенные Токены", если сумма
              компонентов не сходится с `total_token_count`.
            - Это делает отладочный вывод абсолютно прозрачным и самодостаточным.
"""
from typing import Dict, Any, Optional
import re
import html

def escape_markdown_v2(text: str) -> str:
    """
    Экранирует специальные символы для `ParseMode.MARKDOWN_V2`.
    """
    escape_chars = r'\_*[]()~`>#+-=|{}.!'
    return re.sub(f'([{re.escape(escape_chars)}])', r'\\\1', text)

def escape_html(text: str) -> str:
    """
    Экранирует специальные символы HTML: <, > и &.
    """
    return html.escape(text)

def format_stats_text(stats: Dict[str, Any]) -> str:
    """
    Форматирует словарь со статистикой в человекочитаемый текст для MarkdownV2.
    """
    return (
        f"📊 *Статистика бота*\n\n"
        f"👥 Активных чатов: `{stats.get('active_chats', 0)}`\n"
        f"💬 Всего сообщений: `{stats.get('total_messages', 0)}`\n"
        f"🤖 Ответов AI: `{stats.get('ai_responses', 0)}`\n"
        f"⚖️ Срабатываний модерации: `{stats.get('moderation_actions', 0)}`\n"
        f"⚠️ Пользователей с варнами: `{stats.get('users_with_warnings', 0)}`"
    )

def safe_markdown_preview(text: str, length: int) -> str:
    """
    Создает безопасное для MarkdownV2 превью текста.
    """
    if len(text) > length:
        preview = text[:length].rsplit(' ', 1)[0] + "…"
    else:
        preview = text
    return escape_markdown_v2(preview)

def format_debug_info_text(
    usage_metadata: Optional[Dict[str, Any]],
    system_prompt_len: int,
    chat_history_len: int,
    question_len: int
) -> str:
    """
    Форматирует отладочную информацию о запросе в текстовый блок для MarkdownV2.
    Эта функция теперь интеллектуально обрабатывает метаданные токенов.
    """
    lines = [
        escape_markdown_v2("\n\n--- 🐞 DEBUG INFO ---"),
    ]

    if usage_metadata:
        # WHY: Отображаем только те компоненты, которые имеют ненулевое значение,
        #      чтобы не загромождать вывод в чате.
        token_components = {
            "Prompt Tokens": usage_metadata.get("prompt_token_count"),
            "Candidates Tokens": usage_metadata.get("candidates_token_count"),
            "Cached Tokens": usage_metadata.get("cached_content_token_count"),
            "Tool Use Tokens": usage_metadata.get("tool_use_prompt_token_count"),
            "Thinking Tokens": usage_metadata.get("thoughts_token_count"),
        }

        known_sum = 0
        for name, value in token_components.items():
            if value and value > 0:
                lines.append(f"*{escape_markdown_v2(name)}*: `{value}`")
                known_sum += value
        
        total_tokens = usage_metadata.get("total_token_count") or 0
        
        # WHY: Вычисляем сумму известных компонентов и сравниваем с итогом.
        #      Это делает несоответствие, замеченное Архитектором, явным и понятным.
        unaccounted = total_tokens - known_sum
        
        if unaccounted != 0:
            lines.append(f"*_{escape_markdown_v2('-> Неучтенные Токены')}_*: `{unaccounted}`")

        lines.append(f"*{escape_markdown_v2('Total Tokens (API)')}*: `{total_tokens if usage_metadata.get('total_token_count') is not None else 'N/A'}`")

    else:
        lines.append("*Token usage*: `N/A`")
    
    lines.extend([
        escape_markdown_v2("---"),
        f"*System Prompt Chars*: `{system_prompt_len}`",
        f"*Context Chars*: `{chat_history_len}`",
        f"*User Msg Chars*: `{question_len}`",
    ])
    
    return "\n".join(lines)

"""
Модуль для централизованной настройки логирования во всем приложении.

- REFACTOR: Форматтер для `ai_requests.log` значительно упрощен.
            Теперь он ожидает полностью сформированное сообщение, что повышает
            надежность и предотвращает ошибки форматирования.
"""
import logging
import sys
import os
from datetime import datetime
from pathlib import Path

from ..config import DATA_DIR

class TraceIdFilter(logging.Filter):
    """
    Фильтр, который добавляет `trace_id` в каждую запись лога.
    """
    def filter(self, record):
        if not hasattr(record, 'trace_id'):
            record.trace_id = 'system'
        return True

def _cleanup_old_logs(log_dir: Path, pattern: str, max_files: int):
    """Удаляет старые лог-файлы, соответствующие шаблону."""
    try:
        log_files = sorted(log_dir.glob(pattern), key=os.path.getmtime, reverse=True)
        if len(log_files) >= max_files:
            for old_log in log_files[max_files-1:]:
                os.remove(old_log)
    except Exception as e:
        print(f"Warning: Could not remove old log file. Error: {e}")

def setup_logging():
    """
    Настраивает базовую конфигурацию логирования для проекта.
    """
    log_dir = DATA_DIR
    log_dir.mkdir(exist_ok=True)
    
    _cleanup_old_logs(log_dir, "technical_*.log", 10)
    _cleanup_old_logs(log_dir, "ai_requests_*.log", 10)

    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    trace_filter = TraceIdFilter()

    # --- Настройка технического логгера (основного) ---
    tech_log_path = log_dir / f"technical_{timestamp}.log"
    tech_formatter = logging.Formatter("%(asctime)s - [%(trace_id)s] - %(name)s - %(levelname)s - %(message)s")
    
    stream_handler = logging.StreamHandler(sys.stdout)
    stream_handler.setFormatter(tech_formatter)
    stream_handler.addFilter(trace_filter)
    
    tech_file_handler = logging.FileHandler(tech_log_path, encoding='utf-8')
    tech_file_handler.setFormatter(tech_formatter)
    tech_file_handler.addFilter(trace_filter)
    
    # Устанавливаем базовую конфигурацию для нашего приложения на уровне INFO
    logging.basicConfig(level=logging.INFO, handlers=[stream_handler, tech_file_handler], force=True)

    # --- Устранение шума от сторонних библиотек ---
    # WHY: Устанавливаем более высокий уровень логирования для "болтливых" библиотек,
    #      чтобы в технический лог попадали только важные события (предупреждения и ошибки) от них.
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("telegram").setLevel(logging.WARNING)


    # --- Настройка логгера для запросов к ИИ ---
    ai_log_path = log_dir / f"ai_requests_{timestamp}.log"
    # Форматтер максимально прост, т.к. основное форматирование происходит в сервисе.
    ai_formatter = logging.Formatter(
        "%(asctime)s - [%(trace_id)s]\n%(message)s\n"
    )
    
    ai_logger = logging.getLogger('ai_requests')
    ai_logger.setLevel(logging.INFO)
    
    ai_file_handler = logging.FileHandler(ai_log_path, encoding='utf-8')
    ai_file_handler.setFormatter(ai_formatter)
    
    ai_logger.addHandler(ai_file_handler)
    ai_logger.propagate = False

    logger = logging.getLogger(__name__)
    logger.info("Конфигурация логирования настроена. Технический лог: %s, Лог запросов ИИ: %s", tech_log_path, ai_log_path, extra={'trace_id': 'init'})

Это помогает выявить циклические зависимости и синтаксические ошибки
на раннем этапе, не запуская всего бота.
"""

import importlib
import pkgutil
import sys
from pathlib import Path
import pytest

# --- ИСПРАВЛЕНИЕ: Начало ---
# Добавляем корневую директорию проекта в sys.path.
# Это необходимо, чтобы Python мог найти пакет 'bot' при запуске тестов
# из любой директории, включая 'tests/'.
# 1. Получаем путь к текущему файлу (d:/.../tests/test_imports.py)
# 2. Берем его родительскую директорию (.../tests/)
# 3. Берем родительскую директорию еще раз (.../) - это и есть корень проекта.
PROJECT_ROOT = Path(__file__).parent.parent
sys.path.insert(0, str(PROJECT_ROOT))

def _iter_package_modules(package_root: str):
    """
    Рекурсивно обходит все подмодули в указанном пакете.

    Args:
        package_root (str): Имя корневого пакета (например, "bot").

    Yields:
        str: Полное имя каждого найденного модуля (например, "bot.handlers.user").
    """
    try:
        package = importlib.import_module(package_root)
        prefix = package.__name__ + "."
        
        # Используем pkgutil.walk_packages для надежного обхода
        for _, module_name, is_pkg in pkgutil.walk_packages(
            package.__path__, prefix=prefix
        ):
            if not is_pkg:
                yield module_name
    except ImportError as e:
        pytest.fail(f"Не удалось импортировать корневой пакет '{package_root}': {e}")

## 2025-07-30T13:15:00Z - [FIX] - Prevent ConversationHandler State Corruption

**Задача (Task):**
Устранить ошибку, при которой диалоги (`ConversationHandler`) переставали работать в режиме отладки после попытки отредактировать промпт.

**Диагностика (Diagnosis):**
Анализ показал, что при входе в режим редактирования любого промпта (например, "Промпт Санкций") вызывался метод `context.user_data.clear()`. Эта операция полностью очищала словарь `user_data`, удаляя не только данные приложения, но и внутренние ключи, которые библиотека `python-telegram-bot` использует для отслеживания текущего состояния диалога для каждого пользователя. В результате `ConversationHandler` "забывал", что он находится в активном диалоге, и переставал обрабатывать последующие сообщения пользователя. В режиме отладки это приводило к тому, что сообщение перехватывалось глобальным обработчиком и ошибочно классифицировалось как "заметка тестера".

**Решение (Solution):**
1.  **Создан служебный модуль `bot/utils/context_helpers.py`:** В него вынесена новая функция `clear_conversation_user_data`.
2.  **Реализована безопасная очистка:** Эта функция принимает `context` и целенаправленно удаляет только те ключи, которые были установлены логикой нашего приложения (`prompt_type`, `prompt_buffer` и т.д.). Она не затрагивает служебные ключи библиотеки.
3.  **Заменены все вызовы `context.user_data.clear()`:** В файлах `bot/handlers/admin.py` и `bot/handlers/callbacks/prompts.py` все вызовы деструктивной операции `clear()` были заменены на вызов новой, безопасной функции `clear_conversation_user_data`.

**Замысел (Why):**
Это решение продиктовано стремлением к архитектурной чистоте. Создание отдельного служебного файла предотвращает нежелательные зависимости между модулями обработчиков и возможные циклические импорты. Вместо того чтобы полностью стирать контекст, мы теперь аккуратно убираем только то, что необходимо, сохраняя целостность внутреннего состояния `ConversationHandler`. Это гарантирует, что диалоги будут вести себя стабильно и предсказуемо во всех режимах работы, устраняя коренную причину бага.

**Затронутые Файлы (Affected Files):**
*   `data/project_history.md`
*   `bot/utils/context_helpers.py` (новый)
*   `bot/handlers/admin.py`

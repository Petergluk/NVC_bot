import os

ROOT_DIR = Path(__file__).parent.parent
DATA_DIR = ROOT_DIR / "data"
PROMPTS_DIR = ROOT_DIR / "prompts"
PROMPTS_DIR.mkdir(exist_ok=True) # Убедимся, что директория существует

def _load_prompt(file_name: str) -> str:
    """
    Загружает промпт из файла, игнорируя строки-комментарии.
    Если файл не найден или пуст, возвращает пустую строку.
    """
    file_path = PROMPTS_DIR / file_name
    if not file_path.exists():
        logging.warning("Файл промпта не найден: %s. Будет использовано пустое значение.", file_path)
        return ""

    try:
        raw_lines = file_path.read_text(encoding='utf-8').splitlines()
        prompt_lines = [line for line in raw_lines if not line.strip().startswith('#')]
        return "\n".join(prompt_lines).strip()
    except IOError as e:
        logging.error("Не удалось прочитать файл промпта %s: %s. Будет использовано пустое значение.", file_path, e)
        return ""

    def __init__(self):
        self.db_path = DATA_DIR / "bot_database.sqlite3"
        self.conn: Optional[aiosqlite.Connection] = None
        logger.info("Объект Database создан. Путь к БД: %s", self.db_path)

    async def initialize(self):
        try:
            self.conn = await aiosqlite.connect(self.db_path)
            self.conn.row_factory = aiosqlite.Row
            await self._create_tables()
            logger.info("Подключение к базе данных установлено и таблицы проверены.")
        except aiosqlite.Error as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА при инициализации БД: %s", e, exc_info=True)
            raise

    async def _create_tables(self):
        if not self.conn: raise RuntimeError("Подключение к БД не инициализировано.")
        async with self.conn.cursor() as cursor:
            await cursor.execute("CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT NOT NULL)")
            await cursor.execute("""
                CREATE TABLE IF NOT EXISTS messages (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    chat_id INTEGER NOT NULL,
                    message_id INTEGER NOT NULL,
                    user_id INTEGER NOT NULL,
                    user_name TEXT NOT NULL,
                    text TEXT,
                    is_bot BOOLEAN NOT NULL,
                    reply_to_message_id INTEGER,
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
                )
            """)
            await cursor.execute("CREATE TABLE IF NOT EXISTS statistics (key TEXT PRIMARY KEY, value INTEGER NOT NULL)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS user_warnings (user_id INTEGER PRIMARY KEY, count INTEGER NOT NULL DEFAULT 0)")
            await cursor.execute("CREATE TABLE IF NOT EXISTS moderation_triggers (id INTEGER PRIMARY KEY AUTOINCREMENT, trigger_word TEXT NOT NULL UNIQUE, category TEXT NOT NULL DEFAULT 'general')")
            await cursor.execute("CREATE TABLE IF NOT EXISTS moderation_exceptions (id INTEGER PRIMARY KEY AUTOINCREMENT, exception_phrase TEXT NOT NULL UNIQUE, category TEXT NOT NULL DEFAULT 'general')")
            
            await cursor.execute("""
                CREATE TABLE IF NOT EXISTS api_usage_log (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                    model_id TEXT NOT NULL,
                    request_type TEXT NOT NULL,
                    prompt_tokens INTEGER NOT NULL DEFAULT 0,
                    completion_tokens INTEGER NOT NULL DEFAULT 0,
                    total_tokens INTEGER NOT NULL DEFAULT 0
                )
            """)

            await cursor.execute("PRAGMA table_info(messages)")
            columns = [row['name'] for row in await cursor.fetchall()]
            if 'message_id' not in columns:
                await cursor.execute("ALTER TABLE messages ADD COLUMN message_id INTEGER NOT NULL DEFAULT 0")
            if 'user_id' not in columns:
                await cursor.execute("ALTER TABLE messages ADD COLUMN user_id INTEGER NOT NULL DEFAULT 0")
            if 'reply_to_message_id' not in columns:
                await cursor.execute("ALTER TABLE messages ADD COLUMN reply_to_message_id INTEGER")

            await cursor.execute("PRAGMA table_info(moderation_triggers)")
            if 'category' not in [row['name'] for row in await cursor.fetchall()]:
                await cursor.execute("ALTER TABLE moderation_triggers ADD COLUMN category TEXT NOT NULL DEFAULT 'general'")
            
            await cursor.execute("PRAGMA table_info(moderation_exceptions)")
            if 'category' not in [row['name'] for row in await cursor.fetchall()]:
                await cursor.execute("ALTER TABLE moderation_exceptions ADD COLUMN category TEXT NOT NULL DEFAULT 'general'")

        await self.conn.commit()

    async def get_usage_statistics(self) -> Dict[str, Any]:
        """Собирает и агрегирует статистику использования API по разным периодам."""
        if not self.conn:
            return {}

        stats: Dict[str, Any] = {
            'today': defaultdict(lambda: defaultdict(int)),
            'yesterday': defaultdict(lambda: defaultdict(int)),
            'week': defaultdict(lambda: defaultdict(int)),
            'month': defaultdict(lambda: defaultdict(int)),
        }
        
        today_utc = datetime.datetime.now(datetime.timezone.utc).replace(hour=0, minute=0, second=0, microsecond=0)
        
        periods = {
            'today': (today_utc, today_utc + datetime.timedelta(days=1)),
            'yesterday': (today_utc - datetime.timedelta(days=1), today_utc),
            'week': (today_utc - datetime.timedelta(days=today_utc.weekday()), today_utc + datetime.timedelta(days=1)),
            'month': (today_utc.replace(day=1), today_utc + datetime.timedelta(days=1)),
        }
        
        try:
            for period_name, (start_dt, end_dt) in periods.items():
                query = """
                    SELECT
                        model_id,
                        COUNT(*) as request_count,
                        SUM(total_tokens) as total_tokens
                    FROM api_usage_log
                    WHERE timestamp >= ? AND timestamp < ?
                    GROUP BY model_id
                """
                # Конвертируем datetime в строки формата, который точно понимает SQLite
                start_str = start_dt.strftime('%Y-%m-%d %H:%M:%S')
                end_str = end_dt.strftime('%Y-%m-%d %H:%M:%S')

                async with self.conn.execute(query, (start_str, end_str)) as cursor:
                    async for row in cursor:
                        stats[period_name][row['model_id']]['requests'] = row['request_count']
                        stats[period_name][row['model_id']]['tokens'] = row['total_tokens']
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при сборе статистики использования API: %s", e)

        return stats

            await self.conn.execute("INSERT OR REPLACE INTO user_warnings (user_id, count) VALUES (?, ?)", (user_id, count))
            await self.conn.commit()
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при установке предупреждений для user_id %d: %s", user_id, e)
            raise

    async def adjust_warnings(self, user_id: int, amount: int) -> int:
        new_count = max(0, await self.get_warnings(user_id) + amount)
        await self.set_warnings(user_id, new_count)
        return new_count

    async def reset_warnings(self, user_id: int):
        await self.set_warnings(user_id, 0)

    async def get_all_users_with_warnings(self) -> List[Dict]:
        if not self.conn: return []
        try:
            async with self.conn.execute("SELECT user_id, count FROM user_warnings WHERE count > 0 ORDER BY count DESC") as cursor:
                return [dict(row) for row in await cursor.fetchall()]
        except aiosqlite.Error as e:
            logger.error("Ошибка БД при получении списка пользователей с предупреждениями: %s", e)
            return []

        prompt_type = context.user_data.get("prompt_type")
        new_text = context.user_data.get("prompt_buffer", "")
        original_update = context.user_data.get("original_update")

        if not all([prompt_type, new_text, original_update]):
            logger.error("Недостаточно данных для сохранения промпта.", extra={'trace_id': trace_id})
            clear_conversation_user_data(context)
            return

        original_query = original_update.callback_query
        success, error_message = False, ""
        
        save_actions = {
            "facilitator_prompt": (validate_prompt, self.db.set_prompt),
            "moderation_prompt": (validate_prompt, self.db.set_moderation_prompt),
            "moderation_template": (validate_moderation_template, self.db.set_moderation_task_template),
            "welcome_prompt": (lambda p: None if "{user_mention}" in p else "Промпт должен содержать `{user_mention}`.", self.db.set_welcome_prompt)
        }

        validator, saver = save_actions.get(prompt_type, (lambda p: "Неизвестный тип промпта", None))
        error_message = validator(new_text)
        if not error_message and saver:
            await saver(new_text)
            success = True
            logger.info("Промпт '%s' успешно сохранен. Длина: %d", prompt_type, len(new_text), extra={'trace_id': trace_id})

        if success:
            await context.bot.send_message(chat_id=original_query.message.chat_id, text=f"✅ Промпт успешно обновлен\\. Длина: {len(new_text)}\\.")

            "facilitator_prompt": self.callback_handlers.show_facilitator_prompt_menu,
            "moderation_prompt": self.callback_handlers.show_moderation_prompt_menu,
            "moderation_template": self.callback_handlers.show_moderation_template_menu,
            "welcome_prompt": self.callback_handlers.show_welcome_prompt_menu,
        }
        if update_func := menu_update_map.get(prompt_type):
            await update_func(original_update, context)

"""
Базовый класс и общие вспомогательные методы для обработчиков инлайн-кнопок.

Version 1.1.0:
- FIX: Улучшена обработка исключений в `_safe_edit_message` для более
       точного логирования ошибок `Message is too long` и `Message is not modified`.
"""
import logging
from typing import Any

from telegram import CallbackQuery, Message
from telegram.error import BadRequest
from telegram.ext import ContextTypes

from ...database import Database
from ...services.gemini import GeminiService

logger = logging.getLogger(__name__)

class BaseCallbackHandler:
    """Содержит общие методы для всех обработчиков колбэков."""

    def __init__(self, db: Database, gemini: GeminiService) -> None:
        """
        Инициализация базового обработчика.

        Args:
            db (Database): Экземпляр для работы с базой данных.
            gemini (GeminiService): Экземпляр для работы с Gemini API.
        """
        self.db = db
        self.gemini = gemini

    async def _safe_edit_message(self, query: CallbackQuery, text: str, **kw: Any) -> Message | None:
        """Безопасно редактирует сообщение, на которое нажал пользователь."""
        if not query or not query.message:
            logger.warning("Попытка редактировать сообщение без query или query.message.")
            return None
        try:
            # Мы не обрезаем текст здесь, чтобы иметь возможность отловить ошибку о длине
            return await query.message.edit_text(text, **kw)
        except BadRequest as exc:
            # FIX: Детальная обработка исключений BadRequest
            if "Message is too long" in str(exc):
                logger.warning("Не удалось отредактировать сообщение: превышен лимит длины (4096 символов).")
                await self._safe_answer_query(query, "❌ Ошибка: Слишком много текста для отображения.", show_alert=True)
            elif "Message is not modified" in str(exc):
                # Это не ошибка, просто игнорируем
                await self._safe_answer_query(query)
            else:
                logger.error("Telegram BadRequest при редактировании сообщения: %s", exc)
                await self._safe_answer_query(query, "Произошла ошибка при обновлении меню.", show_alert=True)
            return None

    async def _safe_answer_query(self, query: CallbackQuery | None, *args, **kwargs) -> None:
        """Безопасно отвечает на callback query, игнорируя устаревшие."""
        if not query: return
        try:
            await query.answer(*args, **kwargs)
        except BadRequest as e:
            if "Query is too old" in str(e):
                logger.warning("Попытка ответить на устаревший query. Игнорируется.")
            else:
                logger.error("Неожиданная ошибка BadRequest при ответе на query: %s", e)
    
    async def _disable_keyboard(self, query: CallbackQuery) -> None:
        """Безопасно удаляет клавиатуру у сообщения."""
        if not query or not query.message: return
        try:
            await query.message.edit_reply_markup(reply_markup=None)
        except BadRequest as e:
            if "Message is not modified" not in str(e):
                logger.warning("Не удалось убрать клавиатуру: %s", e)

# bot/handlers/callbacks/context.py

"""
Обработчики инлайн-кнопок для меню настроек параметров контекста AI.

Version 1.0.2:
- FIX: Все сообщения переведены на `ParseMode.MARKDOWN_V2`.
- FIX: Все статические и динамические данные теперь корректно экранируются.
"""
import logging
from typing import Dict
from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode
from telegram.ext import ContextTypes
from telegram.error import BadRequest, TelegramError

from ...utils.formatters import escape_markdown_v2
from ..states import FACILITATOR_SETTINGS, CONTEXT_SETTINGS, AWAIT_CONTEXT_VALUE
from .callback_patterns import CallbackPrefix

logger = logging.getLogger(__name__)

class ContextCallbackHandler:
    """Обработчики для ветки /settings -> Настройки контекста."""

    CONTEXT_PARAM_META = {
        "context_messages_limit": {"name": "Контекст диалога", "type": int, "range": (0, 100)},
        "context_judge_limit": {"name": "Контекст для Судьи", "type": int, "range": (0, 50)},
    }

    async def show_context_settings_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Показывает меню настроек лимитов контекста."""
        query = update.callback_query
        if query:
            await self._safe_answer_query(query)

        config = await self.db.get_context_config()
        
        text_lines = [
            "📚 *Настройки Контекста*\n",
            "Здесь настраивается, сколько последних сообщений бот 'помнит' при ответе или модерации\\.",
            "• *Контекст диалога:* Глубина памяти для обычных ответов\\.",
            "• *Контекст для Судьи:* Глубина памяти для анализа нарушений\\.\n"
        ]
        
        buttons = []
        for key, meta in self.CONTEXT_PARAM_META.items():
            value = config.get(key)
            text_lines.append(f"*{escape_markdown_v2(meta['name'])}*: `{value}` сообщ\\.")
            buttons.append(InlineKeyboardButton(f"{meta['name']}", callback_data=f"{CallbackPrefix.CONTEXT_CONFIG_EDIT}{key}"))
        
        kb = [buttons[i:i + 2] for i in range(0, len(buttons), 2)]
        kb.append([InlineKeyboardButton("🔄 Сбросить", callback_data=CallbackPrefix.CONTEXT_CONFIG_RESET), InlineKeyboardButton("⬅️ Назад", callback_data=CallbackPrefix.BACK_TO_FACILITATOR_SETTINGS)])
        
        text = "\n".join(text_lines)
        reply_markup = InlineKeyboardMarkup(kb)
        
        await self._safe_edit_message(query, text, reply_markup=reply_markup, parse_mode=ParseMode.MARKDOWN_V2)
        return CONTEXT_SETTINGS

    async def request_context_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает новое значение для параметра контекста."""
        query = update.callback_query
        await self._safe_answer_query(query)

        param_key = query.data.replace(CallbackPrefix.CONTEXT_CONFIG_EDIT, "")
        if param_key not in self.CONTEXT_PARAM_META:
            return CONTEXT_SETTINGS

        meta = self.CONTEXT_PARAM_META[param_key]
        context.user_data["context_param_key"] = param_key
        context.user_data["original_update"] = update
        
        current_value = (await self.db.get_context_config()).get(param_key)
        min_val, max_val = meta["range"]

        text = (
            f"✏️ *Изменение параметра: {escape_markdown_v2(meta['name'])}*\n\n"
            f"Этот параметр определяет максимальное количество последних сообщений из истории чата, которое будет использоваться\\.\n\n"
            f"*Текущее значение: `{current_value}`*\n"
            f"Введите целое число от `{min_val}` до `{max_val}`\\.\n\n"
            "**Отправьте новое значение или /cancel для отмены\\.**"
        )
        
        await self._safe_edit_message(query, text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        return AWAIT_CONTEXT_VALUE

    async def set_context_value(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Устанавливает новое значение для параметра контекста."""
        msg = update.message
        param_key = context.user_data.get("context_param_key")

        if not param_key or param_key not in self.CONTEXT_PARAM_META:
            await msg.reply_text("❌ Внутренняя ошибка: не удалось определить параметр для изменения\\.")
            context.user_data.clear()
            return FACILITATOR_SETTINGS

        meta = self.CONTEXT_PARAM_META[param_key]
        value_str = msg.text.strip()
        
        try:
            value = meta["type"](value_str)
            min_val, max_val = meta["range"]
            if not (min_val <= value <= max_val):
                raise ValueError(f"Значение должно быть в диапазоне от {min_val} до {max_val}\\.")
        except (ValueError, TypeError) as e:
            await msg.delete()
            original_update = context.user_data.get("original_update")
            if original_update and original_update.callback_query:
                error_text = (
                    f"❌ *Неверное значение для {escape_markdown_v2(meta['name'])}*\n\n"
                    f"Пожалуйста, введите *целое число* от `{meta['range'][0]}` до `{meta['range'][1]}`\\.\n"
                    f"Ошибка: {escape_markdown_v2(str(e))}\n\n"
                    f"**Попробуйте снова или нажмите /cancel\\.**"
                )
                await self._safe_edit_message(original_update.callback_query, error_text, parse_mode=ParseMode.MARKDOWN_V2)
            return AWAIT_CONTEXT_VALUE

        await self.db.set_context_config_value(param_key, value)
        
        try:
            await msg.delete()
        except (BadRequest, TelegramError) as e:
            logger.warning("Не удалось удалить сообщение пользователя при изменении параметра контекста: %s", e)

        original_update = context.user_data.pop("original_update", None)
        context.user_data.clear()
        
        if original_update:
            await self._safe_answer_query(original_update.callback_query, f"✅ {meta['name']} обновлен.")
            return await self.show_context_settings_menu(original_update, context)
        
        return FACILITATOR_SETTINGS

    async def reset_context_config_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Сбрасывает параметры контекста к значениям по умолчанию."""
        query = update.callback_query
        await self._safe_answer_query(query, "Выполняется сброс...")
        
        await self.db.reset_context_config()
        
        await self._safe_answer_query(query, "✅ Параметры контекста сброшены.")
        return await self.show_context_settings_menu(update, context)

            return FACILITATOR_SETTINGS

        meta = self.GENERATION_PARAM_META[param_key]
        value_str = msg.text.strip().replace(',', '.')

            if min_val is not None and value < min_val:
                raise ValueError(f"значение должно быть не меньше {min_val}")

            return AWAIT_GENERATION_VALUE

        await self.db.set_generation_config_value(param_key, value)
        
        try:
            await msg.delete()

            logger.warning("Не удалось удалить сообщение пользователя при изменении параметра: %s", e)

# NVC_bot/bot/handlers/callbacks/items.py
# version: 1.8.0
# 2025-07-30T11:45:00Z
"""
Обработчики инлайн-кнопок для управления триггерами и исключениями модерации.

Version 1.8.0:
- REFACTOR: Методы `request_items_add` и `request_items_delete` были переработаны
            для устранения дублирования кода. Введен новый приватный метод
            `_request_item_modification_input`, который централизует логику
            запроса ввода от пользователя.
Version 1.7.0:
- REFACTOR: Изменена логика `_format_list_for_display` для решения проблемы
            с лимитом на количество "сущностей" (entities) в Telegram.
- FIX: Вместо оборачивания каждого слова в `<code>`, теперь вся строка
       слов для одной категории оборачивается в один тег `<code>`. Это
       радикально снижает количество сущностей и предотвращает сбой форматирования.
"""
from typing import Dict, List

from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode
from telegram.ext import ContextTypes
from telegram.error import BadRequest, TelegramError
import logging

from ...utils.formatters import escape_markdown_v2, escape_html
from ..states import (MANAGE_TRIGGERS, AWAIT_TRIGGERS_TO_ADD, AWAIT_TRIGGERS_TO_DELETE,
                      MANAGE_EXCEPTIONS, AWAIT_EXCEPTIONS_TO_ADD, AWAIT_EXCEPTIONS_TO_DELETE)
from .callback_patterns import CallbackPrefix

logger = logging.getLogger(__name__)

class ItemCallbackHandler:
    """Обработчики для управления списками триггеров и исключений."""

    def _format_list_for_display(self, items: Dict[str, List[str]]) -> str:
        """
        Безопасно форматирует список для отображения в HTML.
        """
        if not items:
            return "Список пуст."
        lines = []
        for category, words in items.items():
            escaped_category = escape_html(category.capitalize())
            lines.append(f"<i>{escaped_category}</i>:")
            
            all_words_string = ", ".join(words)
            escaped_words = escape_html(all_words_string)
            lines.append(f"<code>{escaped_words}</code>")
            
        return "\n\n".join(lines)

    async def _show_item_management_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        query = update.callback_query
        await self._safe_answer_query(query)
        
        items, return_state, callbacks = {}, 0, {}
        if item_type == "triggers":
            items = await self.db.get_all_triggers()
            return_state = MANAGE_TRIGGERS
            callbacks = {"title": "🎯 <b>Управление триггерами</b>", "back": CallbackPrefix.BACK_TO_MODERATION_SETTINGS, "add": CallbackPrefix.TRIGGERS_ADD, "del": CallbackPrefix.TRIGGERS_DELETE, "reset": CallbackPrefix.TRIGGERS_RESET}
        elif item_type == "exceptions":
            items = await self.db.get_all_exceptions()
            return_state = MANAGE_EXCEPTIONS
            callbacks = {"title": "🛡️ <b>Управление исключениями</b>", "back": CallbackPrefix.BACK_TO_MODERATION_SETTINGS, "add": CallbackPrefix.EXCEPTIONS_ADD, "del": CallbackPrefix.EXCEPTIONS_DELETE, "reset": CallbackPrefix.EXCEPTIONS_RESET}
        
        text = f"{callbacks['title']}\n\n{self._format_list_for_display(items)}\n\nДля отмены введите /cancel."
        kb = [
            [InlineKeyboardButton("➕ Добавить", callback_data=callbacks["add"]), InlineKeyboardButton("➖ Удалить", callback_data=callbacks["del"])],
            [InlineKeyboardButton("🔄 Сбросить", callback_data=callbacks["reset"]), InlineKeyboardButton("⬅️ Назад", callback_data=callbacks["back"])]
        ]
        await self._safe_edit_message(query, text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.HTML)
        return return_state

    async def show_triggers_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_item_management_menu(update, context, "triggers")

    async def show_exceptions_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_item_management_menu(update, context, "exceptions")

    async def _request_item_modification_input(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str, action: str) -> int:
        """
        Универсальный метод для запроса ввода от пользователя при добавлении или удалении элементов.

        Args:
            update (Update): Объект обновления Telegram.
            context (ContextTypes.DEFAULT_TYPE): Контекст бота.
            item_type (str): Тип элемента ('triggers' или 'exceptions').
            action (str): Действие ('add' или 'delete').

        Returns:
            int: Следующее состояние для ConversationHandler.
        """
        query = update.callback_query
        await self._safe_answer_query(query)

        # Словарь для хранения текстов и состояний, чтобы избежать дублирования if/else
        config = {
            "triggers": {
                "add": ("Отправьте триггеры для добавления", AWAIT_TRIGGERS_TO_ADD),
                "delete": ("Отправьте триггеры для удаления", AWAIT_TRIGGERS_TO_DELETE),
            },
            "exceptions": {
                "add": ("Отправьте фразы-исключения для добавления", AWAIT_EXCEPTIONS_TO_ADD),
                "delete": ("Отправьте фразы-исключения для удаления", AWAIT_EXCEPTIONS_TO_DELETE),
            }
        }

        base_text, return_state = config[item_type][action]

        # Добавляем детали к базовому тексту
        if action == 'add':
            full_text = f"{base_text}\\.\nФормат: `Категория: слово1, слово2`\nИли просто `слово1, слово2` для категории 'general'\\."
        else:  # 'delete'
            full_text = f"{base_text}, перечислив их через запятую\\."

        context.user_data['original_update'] = update
        await self._safe_edit_message(query, full_text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        
        return return_state

    async def request_items_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает у пользователя элементы (триггеры/исключения) для добавления."""
        item_type = "triggers" if update.callback_query.data == CallbackPrefix.TRIGGERS_ADD else "exceptions"
        return await self._request_item_modification_input(update, context, item_type, "add")

    async def request_items_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        """Запрашивает у пользователя элементы (триггеры/исключения) для удаления."""
        item_type = "triggers" if update.callback_query.data == CallbackPrefix.TRIGGERS_DELETE else "exceptions"
        return await self._request_item_modification_input(update, context, item_type, "delete")

    async def _handle_add_logic(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        msg = update.message
        text = msg.text.strip()
        category = 'general'
        if ':' in text:
            category, text = (part.strip() for part in text.split(':', 1))
            
        items = [item.strip().lower() for item in text.split(',') if item.strip()]
        added_count = 0
        
        add_func = self.db.add_trigger if item_type == "triggers" else self.db.add_exception
        for item in items:
            if await add_func(item, category): added_count += 1
        
        try: await msg.delete()
        except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить сообщение пользователя: %s", e)

        original_update = context.user_data.pop('original_update', None)
        if original_update and original_update.callback_query:
            await self._safe_answer_query(original_update.callback_query, text=f"✅ Добавлено {added_count} из {len(items)} в '{category}'.")
            await self._show_item_management_menu(original_update, context, item_type)
            
        return MANAGE_TRIGGERS if item_type == "triggers" else MANAGE_EXCEPTIONS

    async def _handle_delete_logic(self, update: Update, context: ContextTypes.DEFAULT_TYPE, item_type: str) -> int:
        msg = update.message
        items_to_delete = [item.strip().lower() for item in msg.text.strip().split(',') if item.strip()]
        deleted_count = 0
        
        delete_func = self.db.delete_trigger if item_type == "triggers" else self.db.delete_exception
        for item in items_to_delete:
            if await delete_func(item): deleted_count += 1

        try: await msg.delete()
        except (BadRequest, TelegramError) as e: logger.warning("Не удалось удалить сообщение пользователя: %s", e)

        original_update = context.user_data.pop('original_update', None)
        if original_update and original_update.callback_query:
            await self._safe_answer_query(original_update.callback_query, text=f"✅ Удалено {deleted_count} из {len(items_to_delete)} элементов.")
            await self._show_item_management_menu(original_update, context, item_type)
            
        return MANAGE_TRIGGERS if item_type == "triggers" else MANAGE_EXCEPTIONS

    async def handle_triggers_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_add_logic(update, context, "triggers")

    async def handle_triggers_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_delete_logic(update, context, "triggers")

    async def handle_exceptions_add(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_add_logic(update, context, "exceptions")

    async def handle_exceptions_delete(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._handle_delete_logic(update, context, "exceptions")

    async def reset_items(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        item_type = "triggers" if query.data == CallbackPrefix.TRIGGERS_RESET else "exceptions"
        
        await self._safe_answer_query(query, text="Выполняется сброс...")
        if item_type == "triggers": await self.db.reset_triggers()
        else: await self.db.reset_exceptions()
        
        return await self._show_item_management_menu(update, context, item_type)

"""
Обработчики инлайн-кнопок для управления системными промптами.

Version 2.3.0:
- FEATURE: В подсказку для шаблона сообщения о нарушении добавлена
           переменная `{moderation_rules}`.

Version 2.2.0:
- FIX: Исправлен синтаксис отображения переменных в подсказках.
- FEATURE: В подсказки добавлена переменная `{bot_username}`.
"""
from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.constants import ParseMode, MessageLimit
from telegram.ext import ContextTypes, ConversationHandler

from ...config import settings
from ...utils.formatters import safe_markdown_preview, escape_markdown_v2
from ...utils.context_helpers import clear_conversation_user_data
from ..states import (FACILITATOR_PROMPT_MENU, MODERATION_PROMPT_MENU, MODERATION_TEMPLATE_MENU,
                      WELCOME_PROMPT_MENU, AWAIT_PROMPT_INPUT)
from .callback_patterns import CallbackPrefix

class PromptCallbackHandler:
    """Обработчики для меню управления текстами промптов."""

    async def _get_prompt_data(self, text_type: str) -> tuple:
        if text_type == "facilitator_prompt":
            content = await self.db.get_prompt()
            title = "📜 *Промпт ИИ\\-Фасилитатора*"
            back_cb = CallbackPrefix.BACK_TO_FACILITATOR_SETTINGS
            ret_state = FACILITATOR_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_FACILITATOR_PROMPT
            reset_cb = CallbackPrefix.RESET_FACILITATOR_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "moderation_prompt":
            content = await self.db.get_moderation_prompt()
            title = "🔍 *Промпт Судьи*"
            back_cb = CallbackPrefix.BACK_TO_MODERATION_SETTINGS
            ret_state = MODERATION_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_MODERATION_PROMPT
            reset_cb = CallbackPrefix.RESET_MODERATION_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "moderation_template":
            content = await self.db.get_moderation_task_template()
            title = "📝 *Шаблон Сообщения о Нарушении*"
            back_cb = CallbackPrefix.BACK_TO_MODERATION_SETTINGS
            ret_state = MODERATION_TEMPLATE_MENU
            edit_cb = CallbackPrefix.EDIT_MODERATION_TEMPLATE
            reset_cb = CallbackPrefix.RESET_MODERATION_TEMPLATE
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        elif text_type == "welcome_prompt":
            content = await self.db.get_welcome_prompt()
            title = "📝 *Текст Приветствия*"
            back_cb = CallbackPrefix.BACK_TO_WELCOMES_MENU
            ret_state = WELCOME_PROMPT_MENU
            edit_cb = CallbackPrefix.EDIT_WELCOME_PROMPT
            reset_cb = CallbackPrefix.RESET_WELCOME_PROMPT
            return content, title, back_cb, ret_state, edit_cb, reset_cb
        return None, None, None, None, None, None

    async def _show_text_management_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE, text_type: str) -> int:
        query = update.callback_query
        if query: await self._safe_answer_query(query)
        
        text_content, title, back_cb, ret_state, edit_cb, reset_cb = await self._get_prompt_data(text_type)
        if text_content is None: return ConversationHandler.END

        kb = [[InlineKeyboardButton("✏️ Изменить", callback_data=edit_cb)]]
        preview = safe_markdown_preview(text_content, settings.PROMPT_PREVIEW_LENGTH)
        if len(text_content) > settings.PROMPT_PREVIEW_LENGTH:
            kb.append([InlineKeyboardButton("⬇️ Показать полностью", callback_data=f"{CallbackPrefix.SEND_FULL_PROMPT}{text_type}")])

        kb.extend([[InlineKeyboardButton("🔄 Сбросить", callback_data=reset_cb), InlineKeyboardButton("⬅️ Назад", callback_data=back_cb)]])
        text = f"{title}:\n\n{preview}"
        await self._safe_edit_message(query, text, reply_markup=InlineKeyboardMarkup(kb), parse_mode=ParseMode.MARKDOWN_V2)
        return ret_state

    async def send_full_prompt(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        prompt_type = query.data.replace(CallbackPrefix.SEND_FULL_PROMPT, "")
        text_content, title, _, ret_state, _, _ = await self._get_prompt_data(prompt_type)
        
        if not text_content:
            await query.answer("❌ Не удалось найти текст промпта.", show_alert=True)
            return ret_state

        await query.answer()
        await context.bot.send_message(chat_id=query.message.chat_id, text=f"Полный текст для: {title}", parse_mode=ParseMode.MARKDOWN_V2)
        for i in range(0, len(text_content), MessageLimit.MAX_TEXT_LENGTH):
            await context.bot.send_message(chat_id=query.message.chat_id, text=text_content[i:i + MessageLimit.MAX_TEXT_LENGTH])
        return ret_state

    async def show_facilitator_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "facilitator_prompt")

    async def show_moderation_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "moderation_prompt")
        
    async def show_moderation_template_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "moderation_template")

    async def show_welcome_prompt_menu(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        return await self._show_text_management_menu(update, context, "welcome_prompt")

    async def request_prompt_input(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        await self._safe_answer_query(query)
        clear_conversation_user_data(context)

        cb_data, prompt_type, prompt_text = query.data, "", ""
        
        if cb_data == CallbackPrefix.EDIT_FACILITATOR_PROMPT:
            prompt_type = "facilitator_prompt"
            prompt_text = (
                "Отправьте новый текст для *Промпта ИИ\\-Фасилитатора*\\.\n\n"
                "**Доступные переменные:**\n"
                "`{group_rules}` — описание/правила группы\\.\n"
                "`{bot_username}` — имя пользователя бота \\(например, `@MyBot`\\)\\.\n\n"
                "Для отмены введите /cancel\\."
            )
        elif cb_data == CallbackPrefix.EDIT_MODERATION_PROMPT:
            prompt_type = "moderation_prompt"
            prompt_text = (
                "Отправьте новый текст для *Промпта Судьи*\\.\n\n"
                "**Необязательные переменные:**\n"
                "`{text}` — сообщение пользователя для анализа\\.\n"
                "Для отмены введите /cancel\\."
            )
        elif cb_data == CallbackPrefix.EDIT_MODERATION_TEMPLATE:
            prompt_type = "moderation_template"
            prompt_text = (
                "Отправьте новый текст для *Шаблона Сообщения о Нарушении*\\.\n\n"
                "**Доступные переменные:**\n"
                "`{user_mention}`\n`{violator_message}`\n`{warnings_count}`\n"
                "`{chat_history}`\n`{group_rules}`\n`{moderation_rules}`\n`{bot_username}`\n\n"
                "Для отмены введите /cancel\\."
            )
        elif cb_data == CallbackPrefix.EDIT_WELCOME_PROMPT:
            prompt_type = "welcome_prompt"
            prompt_text = (
                "Отправьте новый текст для *Задачи Приветствия*\\.\n\n"
                "**Доступные переменные:**\n"
                "`{user_mention}` — упоминание нового участника\\.\n"
                "`{group_rules}` — описание/правила группы\\.\n"
                "`{bot_username}` — имя пользователя бота\\.\n\n"
                "Для отмены введите /cancel\\."
            )
        else: return ConversationHandler.END
            
        context.user_data["prompt_type"] = prompt_type
        context.user_data["original_update"] = update
        await self._safe_edit_message(query, prompt_text, reply_markup=None, parse_mode=ParseMode.MARKDOWN_V2)
        return AWAIT_PROMPT_INPUT

    async def reset_prompt_handler(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
        query = update.callback_query
        await self._safe_answer_query(query)
        await self._disable_keyboard(query)
        
        cb_data = query.data
        if cb_data == CallbackPrefix.RESET_FACILITATOR_PROMPT:
            await self.db.reset_prompt()
            await query.answer("✅ Промпт Фасилитатора сброшен.")
            return await self.show_facilitator_prompt_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_MODERATION_PROMPT:
            await self.db.reset_moderation_prompt()
            await query.answer("✅ Промпт Судьи сброшен.")
            return await self.show_moderation_prompt_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_MODERATION_TEMPLATE:
            await self.db.reset_moderation_task_template()
            await query.answer("✅ Шаблон Сообщения сброшен.")
            return await self.show_moderation_template_menu(update, context)
        elif cb_data == CallbackPrefix.RESET_WELCOME_PROMPT:
            await self.db.reset_welcome_prompt()
            await query.answer("✅ Текст Приветствия сброшен.")
            return await self.show_welcome_prompt_menu(update, context)
            
        return ConversationHandler.END

logger = logging.getLogger(__name__)

class UserHandlers:
    """Класс, инкапсулирующий обработчики пользовательских команд."""

    def __init__(self, db: Database, gemini_service: GeminiService, moderation_service: ModerationService):
        self.db = db
        self.gemini = gemini_service
        self.moderation = moderation_service

    async def _process_message(self, message: Message, context: ContextTypes.DEFAULT_TYPE) -> None:
        user, chat, text = message.from_user, message.chat, message.text
        trace_id = str(uuid.uuid4())
        context.user_data['trace_id'] = trace_id

        if not all([user, chat, text]): return

        bot_username = f"@{context.bot.username}"
        is_mention = bot_username.lower() in text.lower()
        is_reply_to_bot = message.reply_to_message and message.reply_to_message.from_user.id == context.bot.id
        should_respond = False
        is_admin_private_chat = chat.type == ChatType.PRIVATE and user.id in ADMIN_IDS_SET
        reply_to_id = message.reply_to_message.message_id if message.reply_to_message else None

        if is_admin_private_chat:
            is_debug_mode = await self.db.is_debug_mode_enabled()
            if is_debug_mode and not (is_mention or is_reply_to_bot):
                logger.info("[ЗАМЕТКА ТЕСТЕРА] Сообщение от админа %s в ЛС: '%s'", user.full_name, text, extra={'trace_id': trace_id})
                await self.db.add_message(chat.id, message.message_id, user.id, user.full_name, text, is_bot=False, reply_to_message_id=reply_to_id)
                return
            else:
                should_respond = True
        
        elif chat.type in [ChatType.GROUP, ChatType.SUPERGROUP]:
            is_violation = await self.moderation.check_message(message, context, trace_id)
            if is_violation:
                await self.moderation.handle_violation(message, context, trace_id)
                return
            await self.db.add_message(chat.id, message.message_id, user.id, user.full_name, text, is_bot=False, reply_to_message_id=reply_to_id)
            if is_mention or is_reply_to_bot:
                should_respond = True
        
        elif chat.type == ChatType.PRIVATE:
            await self.db.add_message(chat.id, message.message_id, user.id, user.full_name, text, is_bot=False, reply_to_message_id=reply_to_id)
            await message.reply_text(PRIVATE_CHAT_REPLY)

        if should_respond:
            await self._respond_to_message(message, context, trace_id, is_admin_private_chat)

    @log_action("new_message")
    async def handle_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        if update.message: await self._process_message(update.message, context)

    @log_action("edited_message")
    async def handle_edited_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        if update.edited_message: await self._process_message(update.edited_message, context)

        if chat.type == ChatType.PRIVATE:

        else:

    @log_action("greet_new_members")
    async def greet_new_members(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
        if not await self.db.is_welcome_enabled() or not update.message or not update.message.new_chat_members: return
        
        chat = update.effective_chat
        is_debug_mode = await self.db.is_debug_mode_enabled()
        
        async def _generate_and_send_welcome(member, chat_description, trace_id):
            """Внутренняя функция для выполнения в фоновой задаче."""
            try:
                # FEATURE: Запись в историю осмысленного системного сообщения
                system_message = f"Пользователь {member.full_name} присоединился к чату."
                await self.db.add_message(
                    chat_id=chat.id,
                    message_id=update.message.message_id,
                    user_id=member.id,
                    user_name="[Системное событие]",
                    text=system_message,

                welcome_template = await self.db.get_welcome_prompt()
                if not welcome_template: return

                formatted_task_content = welcome_template.replace("{user_mention}", member.mention_html())
                formatted_task_content = formatted_task_content.replace("{group_rules}", chat_description)
                formatted_task_content = formatted_task_content.replace("{bot_username}", f"@{context.bot.username}")

                    trace_id=trace_id,
                    request_type="WELCOME",
                    is_debug_mode=is_debug_mode,
                    model_override=model_id,
                    chat_description=chat_description,
                    chat_history=[]
                )

                    sent = await context.bot.send_message(chat.id, response_text, parse_mode=ParseMode.HTML)
                    await self.db.add_message(chat.id, sent.message_id, context.bot.id, "bot", response_text, True)
            except Exception as e:
                logger.error("Ошибка при генерации приветствия в фоновой задаче: %s", e, extra={'trace_id': trace_id}, exc_info=True)

        try:
            chat_description = (await context.bot.get_chat(chat.id)).description or "Правила группы не указаны."
        except Exception as e:
            logger.warning("Не удалось получить описание чата %d: %s", chat.id, e)
            chat_description = "Правила группы не указаны."

        for member in update.message.new_chat_members:
            if member.is_bot: continue
            trace_id = str(uuid.uuid4())
            asyncio.create_task(_generate_and_send_welcome(member, chat_description, trace_id))

    async def _keep_typing_action(self, context: ContextTypes.DEFAULT_TYPE, chat_id: int, trace_id: str) -> None:
        while True:
            try:
                await context.bot.send_chat_action(chat_id=chat_id, action=ChatAction.TYPING)
                await asyncio.sleep(4)
            except asyncio.CancelledError: break
            except Exception as e:
                logger.warning("Ошибка в задаче 'typing': %s", e, extra={'trace_id': trace_id})
                break

    async def _send_long_message(self, context: ContextTypes.DEFAULT_TYPE, chat_id: int, text: str, trace_id: str, **kwargs) -> Optional[Message]:
        if not text:
            logger.warning("Попытка отправить пустое сообщение.", extra={'trace_id': trace_id})
            return None
        sent_message = None
        try:
            for i in range(0, len(text), MessageLimit.MAX_TEXT_LENGTH):
                sent_message = await context.bot.send_message(chat_id=chat_id, text=text[i:i + MessageLimit.MAX_TEXT_LENGTH], **kwargs)
        except BadRequest as e:
            if "Can't parse entities" in str(e):
                logger.warning("Ошибка парсинга Markdown. Отправляю как простой текст. Ошибка: %s", e, extra={'trace_id': trace_id})
                kwargs.pop('parse_mode', None)
                for i in range(0, len(text), MessageLimit.MAX_TEXT_LENGTH):
                    sent_message = await context.bot.send_message(chat_id=chat_id, text=text[i:i + MessageLimit.MAX_TEXT_LENGTH], **kwargs)
            else: raise
        return sent_message

    async def _respond_to_message(self, message: Message, context: ContextTypes.DEFAULT_TYPE, trace_id: str, is_admin_private_chat: bool) -> None:
        user, chat_id, question = message.from_user, message.chat_id, message.text
        typing_task = asyncio.create_task(self._keep_typing_action(context, chat_id, trace_id))
        
        try:
            chat_description = (await context.bot.get_chat(chat_id)).description or "Правила группы не указаны."
        except Exception as e:
            logger.warning("Не удалось получить описание чата %d: %s", chat_id, e, extra={'trace_id': trace_id})
            chat_description = "Правила группы не указаны."

        try:
            context_config = await self.db.get_context_config()
            rich_chat_history = await self.db.get_chat_history(chat_id, context_config["context_messages_limit"])
            is_debug_mode = await self.db.is_debug_mode_enabled()

            response_text, usage_metadata, system_prompt_len = await self.gemini.generate_response(
                question=question,
                chat_history=rich_chat_history,
                trace_id=trace_id,
                request_type="RESPONSE",
                is_debug_mode=is_debug_mode,
                chat_description=chat_description,
                is_admin_chat=is_admin_private_chat
            )
            
            if response_text:
                reply_params = ReplyParameters(message_id=message.message_id)
                sent = await self._send_long_message(context, chat_id, response_text, trace_id, parse_mode=ParseMode.MARKDOWN, reply_parameters=reply_params)
                if sent:
                     await self.db.add_message(chat_id, sent.message_id, context.bot.id, "bot", response_text, True, message.message_id)
                     await self.db.increment_ai_responses()

            if is_debug_mode and is_admin_private_chat:
                debug_info = format_debug_info_text(
                    usage_metadata=usage_metadata,
                    system_prompt_len=system_prompt_len, 
                    chat_history_len=len(json.dumps(rich_chat_history)),
                    question_len=len(question)
                )
                await self._send_long_message(context, chat_id, debug_info, trace_id, parse_mode=ParseMode.MARKDOWN_V2)

        except Exception as e:
            error_message = f"Произошла ошибка. Администраторы уведомлены. (Trace ID: {trace_id})"
            logger.error("Критическая ошибка при генерации ответа: %s", e, exc_info=True, extra={'trace_id': trace_id})
            await message.reply_text(str(e) if isinstance(e, (FailedPrecondition, InternalServerError)) else error_message)
        finally:
            typing_task.cancel()

"""

import traceback

async def error_handler(update: object, context: ContextTypes.DEFAULT_TYPE) -> None:
    trace_id = context.user_data.get('trace_id', str(uuid.uuid4()))
    logger.error("Exception while handling an update:", exc_info=context.error, extra={'trace_id': trace_id})

    if not ADMIN_IDS: return

    tb_list = traceback.format_exception(None, context.error, context.error.__traceback__)
    tb_string = "".join(tb_list)

    update_summary = "N/A"
    if isinstance(update, Update):
        parts = []
        if update.effective_chat: parts.append(f"Chat: {update.effective_chat.title or 'Private'} ({update.effective_chat.id})")
        if update.effective_user: parts.append(f"User: {update.effective_user.full_name} ({update.effective_user.id})")
        if update.callback_query: parts.append(f"Callback: {update.callback_query.data}")
        elif update.effective_message: parts.append(f"Message: {update.effective_message.text[:70] if update.effective_message.text else '<No Text>'}")
        update_summary = " | ".join(parts)

    info_message = (
        f"<b>❗️ Исключение при обработке апдейта</b>\n\n"
        f"<b>Trace ID:</b> <code>{html.escape(trace_id)}</code>\n"
        f"<b>Сводка:</b>\n<pre>{html.escape(update_summary)}</pre>\n\n"
        f"<b>User Data:</b>\n<pre>{html.escape(str(context.user_data))}</pre>"
    )
    
    try:
        await context.bot.send_message(chat_id=ADMIN_IDS[0], text=info_message, parse_mode=ParseMode.HTML)
    except Exception as e:
        logger.error("Не удалось отправить первую часть сообщения об ошибке: %s", e, extra={'trace_id': trace_id})

    traceback_header = "\n\n<pre>------- TRACEBACK -------\n"
    chunk_size = 4096 - len(traceback_header) - len("</pre>")
    
    for i in range(0, len(tb_string), chunk_size):
        chunk = tb_string[i:i+chunk_size]
        message_chunk = f"{traceback_header if i == 0 else '<pre>'}{html.escape(chunk)}</pre>"
        try:
            await context.bot.send_message(chat_id=ADMIN_IDS[0], text=message_chunk, parse_mode=ParseMode.HTML)
        except Exception as e:
            logger.error("Не удалось отправить чанк traceback: %s", e, extra={'trace_id': trace_id})

async def main() -> None:
    """
    Главная асинхронная функция, которая настраивает и запускает бота.
    """
    setup_logging()
    
    db = Database()
    await db.initialize()
    await db.populate_defaults()

    request = HTTPXRequest(connect_timeout=10.0, read_timeout=30.0, write_timeout=30.0)
    app = Application.builder().token(settings.TELEGRAM_BOT_TOKEN).job_queue(None).request(request).build()
    app.add_error_handler(error_handler)

    # REFACTOR: Инъецируем 'app' в сервис для отправки уведомлений
    gemini_service = GeminiService(
        model_names={"pro": settings.GEMINI_PRO_MODEL, "flash": settings.GEMINI_FLASH_MODEL, "lite": settings.GEMINI_LITE_MODEL},
        db=db,
        app=app
    )
    moderation_service = ModerationService(db, gemini_service)

    user_handlers = UserHandlers(db, gemini_service, moderation_service)
    admin_handlers = AdminHandlers(db, gemini_service)
    callback_handlers = CallbackHandlers(db, gemini_service)
    
    admin_handlers.set_callback_handlers(callback_handlers)

    conv_handler = setup_conversations(admin_handlers, callback_handlers)
    app.add_handler(conv_handler)
    
    app.add_handler(CommandHandler("start", user_handlers.start_command))
    app.add_handler(CommandHandler("help", user_handlers.help_command))
    app.add_handler(CommandHandler("stats", admin_handlers.show_stats_command))
    app.add_handler(CommandHandler("warnings", admin_handlers.manage_warnings))
    
    app.add_handler(MessageHandler(filters.StatusUpdate.NEW_CHAT_MEMBERS, user_handlers.greet_new_members))

    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, user_handlers.handle_message))
    app.add_handler(MessageHandler(filters.UpdateType.EDITED_MESSAGE & filters.TEXT & ~filters.COMMAND, user_handlers.handle_edited_message))

    try:
        await app.initialize()
        await app.updater.start_polling()
        await app.start()
        logger.info("Бот запущен и готов к работе. Нажмите Ctrl+C для остановки.", extra={'trace_id': 'init'})
        await asyncio.Event().wait()
    except (KeyboardInterrupt, SystemExit):
        logger.info("Получен сигнал остановки.", extra={'trace_id': 'shutdown'})
    finally:
        logger.info("Начинаю процедуру выключения...", extra={'trace_id': 'shutdown'})
        if app.updater and app.updater.is_running:
            await app.updater.stop()
        if app.running:
            await app.stop()
        await app.shutdown()
        logger.info("Бот успешно остановлен.", extra={'trace_id': 'shutdown'})

logger = logging.getLogger(__name__)
ai_logger = logging.getLogger('ai_requests')

def _handle_api_exceptions(func: Callable[..., Coroutine]) -> Callable[..., Coroutine]:
    """
    Декоратор для обработки исключений API Gemini, включая лимиты.
    Реализует логику повторных попыток и переключения на резервные модели.
    """
    @functools.wraps(func)
    async def wrapper(self: "GeminiService", *args, **kwargs) -> Any:
        max_retries = 3
        base_delay = 5  # секунд

        model_id = kwargs.get('model_override')
        if not model_id:
            if func.__name__ == 'get_moderation_verdict':
                model_id = await self.db.get_moderation_model()
            else:
                model_id = await self.get_active_model_name()
        
        kwargs['model_override'] = model_id

        for attempt in range(max_retries):
            try:
                return await func(self, *args, **kwargs)
            except ResourceExhausted as e:
                trace_id = kwargs.get('trace_id', 'unknown')
                logger.warning(
                    "API Limit Reached (Attempt %d/%d) for model %s. Error: %s",
                    attempt + 1, max_retries, model_id, e, extra={'trace_id': trace_id}
                )

                if "per day" in str(e):
                    fallback_model = MODEL_FALLBACK_CHAIN.get(model_id)
                    tried_models = kwargs.get('_tried_models', set())
                    
                    if fallback_model and fallback_model not in tried_models:
                        await self._notify_admin(
                            f"⚠️ *Лимит API исчерпан*\n\n"
                            f"Суточный лимит для модели `{escape_markdown_v2(model_id)}` достигнут\\. "
                            f"Автоматически переключаюсь на резервную модель: `{escape_markdown_v2(fallback_model)}`\\."
                        )
                        tried_models.add(model_id)
                        kwargs['_tried_models'] = tried_models
                        kwargs['model_override'] = fallback_model
                        model_id = fallback_model
                        continue
                    else:
                        await self._notify_admin(
                            f"🚨 *КРИТИЧЕСКИЙ СБОЙ: Все лимиты исчерпаны*\n\n"
                            f"Суточный лимит для модели `{escape_markdown_v2(model_id)}` достигнут, и резервные модели также недоступны\\. "
                            f"Бот не сможет отвечать до сброса лимитов API\\."
                        )
                        raise e

                elif "per minute" in str(e):
                    if attempt < max_retries - 1:
                        delay = base_delay * (2 ** attempt)
                        logger.info("Retrying in %d seconds...", delay, extra={'trace_id': trace_id})
                        await asyncio.sleep(delay)
                        continue
                    else:
                        logger.error("All retries failed for per-minute limit.", extra={'trace_id': trace_id})
                        raise e
                
                else:
                    raise e
        
        raise RuntimeError("Не удалось получить ответ от API после всех попыток.")

    return wrapper


class GeminiService:
    """
    Инкапсулирует логику работы с моделями Gemini.
    """
    def __init__(self, model_names: Dict[str, str], db: Database, app: Application):
        self.db = db
        self.app = app
        try:
            genai.configure(api_key=settings.GEMINI_API_KEY)
            logger.info("Ключ Gemini API успешно сконфигурирован.", extra={'trace_id': 'init'})
        except Exception as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА: Не удалось сконфигурировать Gemini API.", exc_info=e, extra={'trace_id': 'init'})
            raise

        self.model_names = model_names
        self.models: Dict[str, str] = {model_id: model_id for model_id in model_names.values()}
        logger.info("Модели Gemini зарегистрированы: %s", list(self.models.keys()))

    async def _notify_admin(self, message: str) -> None:
        if not ADMIN_IDS:
            logger.warning("Не удалось отправить уведомление: ADMIN_IDS не задан.")
            return
        try:
            await self.app.bot.send_message(
                chat_id=ADMIN_IDS[0],
                text=message,
                parse_mode=ParseMode.MARKDOWN_V2
            )
        except Exception as e:
            logger.error("Критическая ошибка при отправке уведомления администратору: %s", e)

    async def get_active_model_name(self) -> str:
        return await self.db.get_main_model()

    def get_available_models(self) -> List[str]:
        return list(self.models.keys())

    @_handle_api_exceptions
    async def generate_response(
        self,
        question: str,
        trace_id: str,
        request_type: str,
        is_debug_mode: bool,
        chat_history: Optional[List[Dict]] = None,
        model_override: Optional[str] = None,
        chat_description: Optional[str] = None,
        is_admin_chat: bool = False,
        _tried_models: Optional[Set[str]] = None
    ) -> Tuple[str, Optional[Dict[str, Any]], int]:
        model_id_to_use = model_override
        
        try:
            if is_admin_chat:
                safety_settings = {category: HarmBlockThreshold.BLOCK_NONE for category in HarmCategory if category != HarmCategory.HARM_CATEGORY_UNSPECIFIED}
                safety_level_str = "BLOCK_NONE (Admin Override)"
            else:
                safety_level_str = await self.db.get_safety_level()
                safety_level_enum = getattr(HarmBlockThreshold, safety_level_str, HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE)
                safety_settings = {
                    HarmCategory.HARM_CATEGORY_HARASSMENT: safety_level_enum,
                    HarmCategory.HARM_CATEGORY_HATE_SPEECH: safety_level_enum,
                    HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT: safety_level_enum,
                    HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT: safety_level_enum,
                }
            
            system_prompt = await self.db.get_prompt()
            if chat_description:
                system_prompt = system_prompt.replace("{{group_rules}}", chat_description).replace("{{bot_username}}", f"@{settings.TELEGRAM_BOT_TOKEN.split(':')[0]}")
            system_prompt_len = len(system_prompt)

            model = genai.GenerativeModel(model_name=model_id_to_use, system_instruction=system_prompt, safety_settings=safety_settings)
            
            gen_config_dict = await self.db.get_generation_config()
            generation_config = GenerationConfig(**gen_config_dict)
            
            rich_history = chat_history or []
            api_history = [{"role": "model" if h.get("is_bot") else "user", "parts": [{"text": h.get("text", "")}]} for h in rich_history]
            
            chat_session = model.start_chat(history=api_history)
            content_to_process = question if question.strip() else "Продолжай согласно инструкции."
            
            log_parts = [ f"--- AI REQUEST ({request_type}) ---", f"MODEL: {model_id_to_use}", f"SAFETY: {safety_level_str}" ]
            if is_debug_mode:
                log_parts.extend([
                    f"--- SYSTEM PROMPT ({system_prompt_len} chars) ---\n{system_prompt}",
                    f"--- CHAT HISTORY ({len(json.dumps(rich_history))} chars) ---\n{json.dumps(rich_history, ensure_ascii=False, indent=2)}",
                    f"--- CONTENT ({len(content_to_process)} chars) ---\n{content_to_process}"
                ])
            else:
                 log_parts.extend([
                    f"SYSTEM PROMPT: ({system_prompt_len} chars)",
                    f"CHAT HISTORY: ({len(json.dumps(rich_history))} items)",
                    f"--- CONTENT ({len(content_to_process)} chars) ---\n{truncate_for_log(content_to_process)}"
                 ])
            log_parts.append("--- END OF PAYLOAD ---")
            ai_logger.info("\n".join(log_parts), extra={'trace_id': trace_id})

            response = await chat_session.send_message_async(content=content_to_process, generation_config=generation_config)
            
            if not response.parts:
                finish_reason = response.candidates[0].finish_reason if response.candidates else 'UNKNOWN'
                logger.warning("Модель не вернула контент. Причина: %s.", finish_reason, extra={'trace_id': trace_id})
                return "К сожалению, я не могу ответить на это. Возможно, в запросе или контексте содержатся неоднозначные выражения.", None, system_prompt_len

            usage_metadata_obj = getattr(response, 'usage_metadata', {})
            full_usage_metadata = { "prompt_token_count": getattr(usage_metadata_obj, "prompt_token_count", 0), "candidates_token_count": getattr(usage_metadata_obj, "candidates_token_count", 0), "total_token_count": getattr(usage_metadata_obj, "total_token_count", 0), }
            
            if full_usage_metadata.get("total_token_count", 0) > 0:
                await self.db.log_api_usage(model_id_to_use, request_type, full_usage_metadata)

            ai_logger.info(f"--- USAGE INFO ---\n{json.dumps(full_usage_metadata, indent=2)}\n--- END REQUEST ---", extra={'trace_id': trace_id})

            return response.text, full_usage_metadata, system_prompt_len

        except FailedPrecondition as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА: API отклоняет запросы (геоблокировка?). Ошибка: %s", e, exc_info=True, extra={'trace_id': trace_id})
            raise FailedPrecondition("К сожалению, сервис временно недоступен. Возможно, это связано с региональными ограничениями (попробуйте использовать VPN).") from e
        except InternalServerError as e:
            logger.error("КРИТИЧЕСКАЯ ОШИБКА: Внутренняя ошибка сервера Google API. Ошибка: %s", e, exc_info=True, extra={'trace_id': trace_id})
            raise InternalServerError("Произошла внутренняя ошибка на стороне AI. Пожалуйста, попробуйте повторить ваш запрос через некоторое время.") from e
        except Exception as e:
            if not isinstance(e, ResourceExhausted):
                logger.error("Ошибка при генерации ответа от Gemini: %s", e, exc_info=True, extra={'trace_id': trace_id})
                raise Exception(f"Произошла ошибка при обращении к AI. (Тип: {type(e).__name__})") from e
            raise e

    @_handle_api_exceptions
    async def get_moderation_verdict(
        self,
        text: str,
        prompt_template: str,
        trace_id: str,
        is_debug_mode: bool,
        chat_history: str = "",
        *,
        model_override: Optional[str] = None,
        _tried_models: Optional[Set[str]] = None
    ) -> bool:
        model_id_to_use = model_override
        
        safety_settings = {
            HarmCategory.HARM_CATEGORY_HARASSMENT: HarmBlockThreshold.BLOCK_NONE,
            HarmCategory.HARM_CATEGORY_HATE_SPEECH: HarmBlockThreshold.BLOCK_NONE,
            HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT: HarmBlockThreshold.BLOCK_NONE,
            HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT: HarmBlockThreshold.BLOCK_NONE,
        }
        
        model = genai.GenerativeModel(model_id_to_use, safety_settings=safety_settings)
        final_prompt = prompt_template.format(text=text, chat_history=chat_history)
        
        try:
            log_parts = [ f"--- AI REQUEST (MODERATION_VERDICT) ---", f"MODEL: {model_id_to_use}" ]
            if is_debug_mode:
                log_parts.extend([ f"--- TASK PROMPT ({len(prompt_template)} chars) ---\n{prompt_template}", f"--- CHAT HISTORY ({len(chat_history)} chars) ---\n{chat_history}", f"--- CONTENT ({len(text)} chars) ---\n{text}", ])
            else:
                log_parts.extend([ f"TASK PROMPT: ({len(prompt_template)} chars)", f"CHAT HISTORY: ({len(chat_history)} chars)", f"--- CONTENT ({len(text)} chars) ---\n{truncate_for_log(text)}" ])
            log_parts.append("--- END OF PAYLOAD ---")
            ai_logger.info("\n".join(log_parts), extra={'trace_id': trace_id})

            response = await model.generate_content_async([final_prompt], generation_config=GenerationConfig(temperature=0.0, response_mime_type="application/json"))
            
            usage_metadata_obj = getattr(response, 'usage_metadata', {})
            full_usage_metadata = { "prompt_token_count": getattr(usage_metadata_obj, "prompt_token_count", 0), "candidates_token_count": getattr(usage_metadata_obj, "candidates_token_count", 0), "total_token_count": getattr(usage_metadata_obj, "total_token_count", 0), }

            if full_usage_metadata.get("total_token_count", 0) > 0:
                await self.db.log_api_usage(model_id_to_use, "MODERATION_VERDICT", full_usage_metadata)

            ai_logger.info(f"--- USAGE INFO ---\n{json.dumps(full_usage_metadata, indent=2)}\n--- END REQUEST ---", extra={'trace_id': trace_id})

            if not response.parts:
                logger.warning("Модель модерации не вернула контент. Finish reason: %s.", response.candidates[0].finish_reason, extra={'trace_id': trace_id})
                return False

            verdict_data = json.loads(re.sub(r'```json\n?|```', '', response.text, flags=re.DOTALL))
            ai_logger.info(f"--- MODERATION RESPONSE (JSON) ---\n{json.dumps(verdict_data, indent=2, ensure_ascii=False)}\n--- END RESPONSE ---", extra={'trace_id': trace_id})
            return verdict_data.get("is_violation") is True

        except (FailedPrecondition, InternalServerError) as e:
            logger.critical("КРИТИЧЕСКАЯ ОШИБКА: API модерации недоступно. Ошибка: %s", e, exc_info=True, extra={'trace_id': trace_id})
            return False
        except Exception as e:
            if not isinstance(e, ResourceExhausted):
                logger.error("Ошибка при получении вердикта модерации: %s", e, exc_info=True, extra={'trace_id': trace_id})
            return False

"""
Сервис, отвечающий за модерацию сообщений в чате.

Version 4.4.0:
- FIX: Убрано некорректное экранирование Markdown-символов
       при подготовке данных для AI, что искажало смысл
       передаваемого контекста.

Version 4.3.0:
- FIX: Убран аргумент `model_id` из вызова `gemini.get_moderation_verdict`.
"""
import logging
import re
from datetime import datetime, timedelta
from typing import Dict, List, Any
from telegram import Message
from telegram.ext import ContextTypes
from telegram.error import TelegramError, BadRequest
from telegram.constants import ParseMode

from ..database import Database
from .gemini import GeminiService
from ..utils.formatters import escape_markdown_v2

logger = logging.getLogger(__name__)

class ModerationService:
    """
    Инкапсулирует логику проверки сообщений и применения санкций.
    """

    def __init__(self, db: Database, gemini: GeminiService):
        self.db = db
        self.gemini = gemini

    async def check_message(self, message: Message, context: ContextTypes.DEFAULT_TYPE, trace_id: str) -> bool:
        if not await self.db.is_moderation_enabled(): return False
        user, chat, text = message.from_user, message.chat, message.text
        if not all([user, chat, text]): return False
        
        try:
            if user.id in {admin.user.id for admin in await context.bot.get_chat_administrators(chat.id)}:
                return False
        except TelegramError as e:
            logger.warning("Не удалось получить список админов чата %d: %s", chat.id, e, extra={'trace_id': trace_id})

        text_lower = text.lower()
        triggers = await self.db.get_all_triggers_flat()
        exceptions = await self.db.get_all_exceptions_flat()

        if not any(re.search(r'\b' + re.escape(trigger) + r'\b', text_lower, re.IGNORECASE) for trigger in triggers):
            return False
        if any(exc in text_lower for exc in exceptions):
            return False
        
        logger.info("Сообщение передается AI-судье.", extra={'trace_id': trace_id})
        
        context_config = await self.db.get_context_config()
        history_list = await self.db.get_chat_history(chat.id, context_config["context_judge_limit"])
        history_str = self._format_history_for_template(history_list)
        prompt_template = await self.db.get_moderation_prompt()
        is_debug_mode = await self.db.is_debug_mode_enabled()
        
        return await self.gemini.get_moderation_verdict(text, prompt_template, trace_id, is_debug_mode, history_str)

    def _format_history_for_template(self, history: List[Dict[str, Any]]) -> str:
        if not history: return "Контекст диалога пуст."
        messages_by_id = {msg['message_id']: msg for msg in history}
        lines = []
        for msg in history:
            line = f"{msg.get('user_name', 'bot')}:"
            if reply_id := msg.get('reply_to_message_id'):
                if reply_msg := messages_by_id.get(reply_id):
                    line += f" (в ответ на «{(reply_msg.get('text') or '')[:50]}...» от {reply_msg.get('user_name', 'unknown')})"
            line += f"\n> {msg.get('text', '')}"
            lines.append(line)
        return "\n\n".join(lines)

    def _format_sanction_rules(self, rules: List[Dict]) -> str:
        """
        Форматирует правила санкций в человекочитаемый Markdown-список.
        """
        if not rules:
            return "Правила санкций не настроены."

        def get_duration_text(hours: int) -> str:
            if hours == 1: return "1 час"
            if 2 <= hours <= 4: return f"{hours} часа"
            if hours == 24: return "1 сутки"
            if hours == 168: return "7 суток (неделю)"
            return f"{hours} часов"

        lines = ["Действующие в чате правила применения санкций:"]
        for rule in sorted(rules, key=lambda x: x['warnings']):
            warnings_count = rule['warnings']
            action = rule['action']
            
            if action == 'BAN':
                action_text = "полная блокировка в чате (бан)"
            elif action == 'MUTE' and 'duration_hours' in rule and rule['duration_hours']:
                duration_text = get_duration_text(rule['duration_hours'])
                action_text = f"временное ограничение на отправку сообщений (мьют) на {duration_text}"
            else:
                continue
            lines.append(f"• Накопление {warnings_count} предупреждений приводит к: {action_text}.")
        return "\n".join(lines)

    async def handle_violation(self, message: Message, context: ContextTypes.DEFAULT_TYPE, trace_id: str) -> None:
        user, chat, text = message.from_user, message.chat, message.text
        if not all([user, chat, text]): return

        await self.db.increment_moderation_actions()
        new_warnings_count = await self.db.adjust_warnings(user.id, 1)
        logger.info("Пользователю %s (%s) выдано предупреждение. Всего: %d", user.full_name, user.id, new_warnings_count, extra={'trace_id': trace_id})
        
        sanction_rules = await self.db.get_sanction_rules()
        applied_sanction = next((r for r in sorted(sanction_rules, key=lambda x: x['warnings'], reverse=True) if new_warnings_count >= r['warnings']), None)

        try:
            chat_description = (await context.bot.get_chat(chat.id)).description or "Правила группы не указаны."
        except Exception as e:
            logger.warning("Не удалось получить описание чата %d: %s", chat.id, e, extra={'trace_id': trace_id})
            chat_description = "Правила группы не указаны."

        task_template = await self.db.get_moderation_task_template()
        context_config = await self.db.get_context_config()
        history_list = await self.db.get_chat_history(chat.id, context_config["context_judge_limit"])
        
        # FIX: Убрано экранирование для переменных, отправляемых в AI.
        # Они должны быть в "сыром" виде.
        formatted_task_content = task_template.replace('{user_mention}', user.mention_markdown_v2()) # Этот уже экранирован
        formatted_task_content = formatted_task_content.replace('{violator_message}', text) # Без экранирования
        formatted_task_content = formatted_task_content.replace('{warnings_count}', str(new_warnings_count))
        formatted_task_content = formatted_task_content.replace('{chat_history}', self._format_history_for_template(history_list)) # Без экранирования
        formatted_task_content = formatted_task_content.replace('{group_rules}', chat_description) # Без экранирования
        formatted_task_content = formatted_task_content.replace('{moderation_rules}', self._format_sanction_rules(sanction_rules)) # Без экранирования
        formatted_task_content = formatted_task_content.replace('{bot_username}', f'@{context.bot.username}')
        
        is_debug_mode = await self.db.is_debug_mode_enabled()
        ai_response = ""
        
        try:
            ai_response, _, _ = await self.gemini.generate_response(
                question=formatted_task_content,
                chat_history=history_list,
                trace_id=trace_id,
                request_type="VIOLATION_MESSAGE",
                is_debug_mode=is_debug_mode
            )
        except Exception as e:
            logger.error("Ошибка при генерации ответа AI-фасилитатором: %s", e, exc_info=True, extra={'trace_id': trace_id})
            fallback_sanction = "Применена санкция в соответствии с правилами." if applied_sanction else ""
            ai_response = f"⚠️ {user.mention_markdown_v2()}, ваше сообщение было расценено как оценочное суждение. Это ваше *{new_warnings_count}*-е предупреждение. {fallback_sanction}"

        if ai_response:
            try:
                # Отправляем ответ с parse_mode=Markdown, т.к. AI должен генерировать валидную разметку
                await message.reply_text(text=ai_response, parse_mode=ParseMode.MARKDOWN)
            except BadRequest as e:
                if "Can't parse entities" in str(e):
                    # Если AI вернул некорректный Markdown, отправляем как простой текст
                    await message.reply_text(text=re.sub(r'[*_`\[\]()~>#+\-=|{}.!]', '', ai_response))
                else: 
                    logger.error("Неожиданная ошибка BadRequest: %s", e, extra={'trace_id': trace_id})

        if applied_sanction:
            action, duration = applied_sanction['action'], applied_sanction.get('duration_hours')
            try:
                if action == 'MUTE' and duration:
                    await context.bot.restrict_chat_member(chat.id, user.id, permissions={'can_send_messages': False}, until_date=datetime.now() + timedelta(hours=duration))
                    logger.info("Пользователь %s замьючен на %d часов.", user.id, duration, extra={'trace_id': trace_id})
                elif action == 'BAN':
                    await context.bot.ban_chat_member(chat.id, user.id)
                    logger.info("Пользователь %s забанен.", user.id, extra={'trace_id': trace_id})
            except Exception as e:
                logger.error("Не удалось применить санкцию '%s' для %s: %s", action, user.id, e, extra={'trace_id': trace_id})

"""
Вспомогательные функции для работы с контекстом ConversationHandler.

Этот модуль предоставляет безопасные способы управления данными в `context.user_data`,
чтобы избежать случайного удаления внутренних ключей, используемых библиотекой
python-telegram-bot для отслеживания состояния диалога.

Зачем это нужно:
Прямой вызов `context.user_data.clear()` уничтожает ВСЕ данные, включая
служебные ключи, которые `ConversationHandler` использует для ведения
диалога. Это приводит к "амнезии" обработчика и непредсказуемому поведению.
Эта функция решает проблему, удаляя только те ключи, которые были явно
добавлены логикой нашего приложения.
"""
from telegram.ext import ContextTypes
import logging

logger = logging.getLogger(__name__)

def clear_conversation_user_data(context: ContextTypes.DEFAULT_TYPE) -> None:
    """
    Безопасно очищает только те ключи из user_data, которые были установлены
    в ходе диалогов настроек, не затрагивая внутренние ключи PTB.

    Args:
        context (ContextTypes.DEFAULT_TYPE): Контекст бота, содержащий user_data.
    """
    # Список ключей, которые устанавливаются логикой нашего приложения в user_data
    # и должны быть очищены по завершении или отмене диалога.
    app_keys_to_clear = [
        "prompt_type",
        "prompt_buffer",
        "original_update",
        "prompt_commit_task",
        "context_param_key",
        "gen_param_key",
        "current_sanction",
        "trace_id",
    ]
    
    logger.debug("Очистка пользовательских ключей из user_data: %s", app_keys_to_clear)
    
    for key in app_keys_to_clear:
        if key in context.user_data:
            try:
                del context.user_data[key]
            except KeyError:
                # Это может произойти в редких случаях (например, при гонке состояний),
                # но это безопасно проигнорировать.
                pass

"""
Модуль с декораторами для логирования, проверки прав и обратной связи.
"""
import functools
import logging
from typing import Callable, Any
from telegram import Update
from telegram.ext import ContextTypes
from telegram.constants import ChatAction

# --- ИСПРАВЛЕНИЕ: Импортируем уже обработанный список ADMIN_IDS ---
from ..config import ADMIN_IDS

logger = logging.getLogger(__name__)

# --- ИСПРАВЛЕНИЕ: Преобразуем готовый список ID в множество для быстрой проверки ---
# Логика парсинга и обработки ошибок теперь централизована в config.py.
ADMIN_IDS_SET = set(ADMIN_IDS)
if not ADMIN_IDS_SET:
    logger.warning("Список ADMIN_IDS пуст или не задан в .env. "
                   "Административные команды будут недоступны.")


def admin_only(func: Callable) -> Callable:
    """
    Декоратор, который проверяет, является ли пользователь администратором.
    Разрешает выполнение функции только для пользователей, чьи ID есть в ADMIN_IDS_SET.
    """
    @functools.wraps(func)
    async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
        """Внутренняя обертка, выполняющая проверку."""
        if not update.effective_user:
            logger.warning("Не удалось определить пользователя для проверки прав доступа.")
            return None
        
        user_id = update.effective_user.id
        
        # --- ИСПРАВЛЕНИЕ: Используем ADMIN_IDS_SET для проверки ---
        if user_id not in ADMIN_IDS_SET:
            logger.warning("Пользователь %d попытался выполнить команду только для администраторов: %s",
                         user_id, func.__name__)
            # Проверяем, есть ли у сообщения атрибут message, чтобы ответить
            if update.message:
                await update.message.reply_text("⛔ У вас нет прав для выполнения этой команды.")
            return None
            
        return await func(self, update, context, *args, **kwargs)
        
    return wrapper


def log_action(action_name: str) -> Callable:
    """
    Декоратор для логирования действий пользователя.
    
    Args:
        action_name (str): Название действия для отображения в логе.
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
            """Внутренняя обертка, выполняющая логирование."""
            user = update.effective_user
            chat = update.effective_chat
            
            if user and chat:
                log_message = (
                    f"Действие '{action_name}' вызвано пользователем {user.full_name} "
                    f"(ID: {user.id}, Username: @{user.username}) "
                    f"в чате '{getattr(chat, 'title', 'личный чат')}' (ID: {chat.id}, Type: {chat.type})."
                )
                logger.info(log_message)
            else:
                logger.info("Действие '%s' вызвано без определения пользователя/чата.", action_name)

            return await func(self, update, context, *args, **kwargs)
            
        return wrapper
    return decorator


def typing_action(func: Callable) -> Callable:
    """
    Декоратор, который отправляет действие 'typing...' в чат.
    Это дает пользователю понять, что бот обрабатывает запрос.
    """
    @functools.wraps(func)
    async def wrapper(self: Any, update: Update, context: ContextTypes.DEFAULT_TYPE, *args, **kwargs) -> Any:
        """Внутренняя обертка, отправляющая действие."""
        if update.effective_chat:
            # Отправляем действие "печатает..." перед выполнением основной логики
            await context.bot.send_chat_action(
                chat_id=update.effective_chat.id,
                action=ChatAction.TYPING
            )
        # Выполняем оригинальную функцию
        return await func(self, update, context, *args, **kwargs)
        
    return wrapper

from typing import Dict, Any, Optional
import re
import html
import datetime

from ..config import MODEL_NAME_MAP

def escape_markdown_v2(text: str) -> str:
    escape_chars = r'\_*[]()~`>#+-=|{}.!'
    return re.sub(f'([{re.escape(escape_chars)}])', r'\\\1', text)

def escape_html(text: str) -> str:
    return html.escape(text)

def truncate_for_log(text: str, head: int = 250, tail: int = 150) -> str:
    if not isinstance(text, str): text = str(text)
    if len(text) > head + tail + 20:
        return f"{text[:head]}\n... (усечено, всего {len(text)} симв.) ...\n{text[-tail:]}"
    return text

def format_stats_text(stats: Dict[str, Any]) -> str:
    """Форматирует расширенную статистику, включая использование API."""

    def _format_period_stats(period_data: Dict[str, Dict[str, int]]) -> str:
        if not period_data:
            return "_Нет данных за этот период\\._\n"
        
        lines = []
        sorted_models = sorted(period_data.items(), key=lambda item: item[1].get('requests', 0), reverse=True)
        
        for model_id, data in sorted_models:
            model_name = MODEL_NAME_MAP.get(model_id, model_id)
            requests = data.get('requests', 0)
            tokens = data.get('tokens', 0)
            lines.append(
                f"  `•` *{escape_markdown_v2(model_name)}*:\n"
                f"    Запросов: `{requests}` ⎥ Токенов: `{tokens:,}`"
            )
        return "\n".join(lines)

    lines = [
        f"📊 *Общая статистика бота*",
        f"👥 Активных чатов: `{stats.get('active_chats', 0)}`",
        f"💬 Всего сообщений в БД: `{stats.get('total_messages', 0)}`",
        f"🤖 Ответов AI \\(счетчик\\): `{stats.get('ai_responses', 0)}`",
        f"⚖️ Срабатываний модерации: `{stats.get('moderation_actions', 0)}`",
        f"⚠️ Пользователей с варнами: `{stats.get('users_with_warnings', 0)}`",
        "\n" + escape_markdown_v2("---") + "\n",
        "📈 *Статистика использования API*",
    ]

    usage = stats.get("usage", {})
    
    lines.append("\n🗓️ *Сегодня*")
    lines.append(_format_period_stats(usage.get('today', {})))
    
    lines.append("\n🗓️ *Вчера*")
    lines.append(_format_period_stats(usage.get('yesterday', {})))

    today_utc = datetime.datetime.now(datetime.UTC).date()
    
    days_in_week = today_utc.weekday() + 1
    week_stats = usage.get('week', {})
    total_week_req = sum(d.get('requests', 0) for d in week_stats.values())
    total_week_tok = sum(d.get('tokens', 0) for d in week_stats.values())
    avg_week_req = total_week_req / days_in_week if days_in_week > 0 else 0
    avg_week_tok = total_week_tok / days_in_week if days_in_week > 0 else 0
    lines.append(f"\n🗓️ *Среднее за неделю \\(в день, за {days_in_week}/7 дн\\.\\)*")
    lines.append(f"  Запросов: `{avg_week_req:.1f}` ⎥ Токенов: `{int(avg_week_tok):,}`")

    days_in_month = today_utc.day
    month_stats = usage.get('month', {})
    total_month_req = sum(d.get('requests', 0) for d in month_stats.values())
    total_month_tok = sum(d.get('tokens', 0) for d in month_stats.values())
    avg_month_req = total_month_req / days_in_month if days_in_month > 0 else 0
    avg_month_tok = total_month_tok / days_in_month if days_in_month > 0 else 0
    lines.append(f"\n🗓️ *Среднее за месяц \\(в день, за {days_in_month} дн\\.\\)*")
    lines.append(f"  Запросов: `{avg_month_req:.1f}` ⎥ Токенов: `{int(avg_month_tok):,}`")

    return "\n".join(lines)


def safe_markdown_preview(text: str, length: int) -> str:
    preview = text[:length].rsplit(' ', 1)[0] + "…" if len(text) > length else text
    return escape_markdown_v2(preview)

def format_debug_info_text(
    usage_metadata: Optional[Dict[str, Any]],
    system_prompt_len: int,
    chat_history_len: int,
    question_len: int
) -> str:
    lines = [escape_markdown_v2("\n\n--- 🐞 DEBUG INFO ---")]

    if usage_metadata:
        token_lines = []
        prompt_tokens = usage_metadata.get("prompt_token_count", 0)
        candidates_tokens = usage_metadata.get("candidates_token_count", 0)
        total_tokens = usage_metadata.get("total_token_count", 0)

        if prompt_tokens > 0: token_lines.append(f"*Prompt Tokens*: `{prompt_tokens}`")
        if candidates_tokens > 0: token_lines.append(f"*Candidates Tokens*: `{candidates_tokens}`")
        if total_tokens > 0: token_lines.append(f"*Total Tokens \\(API\\)*: `{total_tokens}`")
        
        if not token_lines: lines.append("*Token usage*: `N/A`")
        else: lines.extend(token_lines)
    else:
        lines.append("*Token usage*: `N/A`")
    
    lines.extend([
        escape_markdown_v2("---"),
        f"*System Prompt Chars*: `{system_prompt_len}`",
        f"*Context Chars*: `{chat_history_len}`",
        f"*User Msg Chars*: `{question_len}`",
    ])
    
    return "\n".join(lines)

"""

- REFACTOR: Удалена устаревшая функция `_cleanup_old_logs`, так как ее
            функциональность теперь покрывается `RotatingFileHandler`.

Version 3.3.0:
- REFACTOR: Установлен уровень логирования WARNING для `httpx` и `telegram`.
- REFACTOR: Форматтер для `ai_requests.log` упрощен.
"""
import logging
import sys
from logging.handlers import RotatingFileHandler
from datetime import datetime
from pathlib import Path

from ..config import DATA_DIR

class TraceIdFilter(logging.Filter):
    """
    Фильтр, который добавляет `trace_id` в каждую запись лога.
    """
    def filter(self, record):
        if not hasattr(record, 'trace_id'):
            record.trace_id = 'system'
        return True

def setup_logging():
    """
    Настраивает базовую конфигурацию логирования для проекта.
    """
    log_dir = DATA_DIR
    log_dir.mkdir(exist_ok=True)
    
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    trace_filter = TraceIdFilter()

    # --- Настройка технического логгера (основного) ---
    tech_log_path = log_dir / f"technical_{timestamp}.log"
    tech_formatter = logging.Formatter("%(asctime)s - [%(trace_id)s] - %(name)s - %(levelname)s - %(message)s")
    
    stream_handler = logging.StreamHandler(sys.stdout)
    stream_handler.setFormatter(tech_formatter)
    stream_handler.addFilter(trace_filter)
    
    # FEATURE: Используем ротацию логов по размеру. 2 МБ на файл, храним 5 старых.
    tech_file_handler = RotatingFileHandler(
        tech_log_path, maxBytes=2*1024*1024, backupCount=5, encoding='utf-8'
    )
    tech_file_handler.setFormatter(tech_formatter)
    tech_file_handler.addFilter(trace_filter)
    
    logging.basicConfig(level=logging.INFO, handlers=[stream_handler, tech_file_handler], force=True)

    # --- Устранение шума от сторонних библиотек ---
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("telegram").setLevel(logging.WARNING)

    # --- Настройка логгера для запросов к ИИ ---
    ai_log_path = log_dir / f"ai_requests_{timestamp}.log"
    ai_formatter = logging.Formatter(
        "%(asctime)s - [%(trace_id)s]\n%(message)s\n"
    )
    
    ai_logger = logging.getLogger('ai_requests')
    ai_logger.setLevel(logging.INFO)
    
    # FEATURE: Используем ротацию логов по размеру. 2 МБ на файл, храним 5 старых.
    ai_file_handler = RotatingFileHandler(
        ai_log_path, maxBytes=2*1024*1024, backupCount=5, encoding='utf-8'
    )
    ai_file_handler.setFormatter(ai_formatter)
    
    ai_logger.addHandler(ai_file_handler)
    ai_logger.propagate = False

    logger = logging.getLogger(__name__)
    logger.info("Конфигурация логирования настроена. Технический лог: %s, Лог запросов ИИ: %s", tech_log_path, ai_log_path, extra={'trace_id': 'init'})

        "{chat_history}",

    return None

"""
Тест для проверки корректности импортов всех модулей в проекте.

Это помогает выявить циклические зависимости и синтаксические ошибки
на раннем этапе, не запуская всего бота.
"""

import importlib
import pkgutil
import sys
from pathlib import Path
import pytest

# --- ИСПРАВЛЕНИЕ: Начало ---
# Добавляем корневую директорию проекта в sys.path.
# Это необходимо, чтобы Python мог найти пакет 'bot' при запуске тестов
# из любой директории, включая 'tests/'.
# 1. Получаем путь к текущему файлу (d:/.../tests/test_imports.py)
# 2. Берем его родительскую директорию (.../tests/)
# 3. Берем родительскую директорию еще раз (.../) - это и есть корень проекта.
PROJECT_ROOT = Path(__file__).parent.parent
sys.path.insert(0, str(PROJECT_ROOT))
# --- ИСПРАВЛЕНИЕ: Конец ---


def _iter_package_modules(package_root: str):
    """
    Рекурсивно обходит все подмодули в указанном пакете.

    Args:
        package_root (str): Имя корневого пакета (например, "bot").

    Yields:
        str: Полное имя каждого найденного модуля (например, "bot.handlers.user").
    """
    try:
        package = importlib.import_module(package_root)
        prefix = package.__name__ + "."
        
        # Используем pkgutil.walk_packages для надежного обхода
        for _, module_name, is_pkg in pkgutil.walk_packages(
            package.__path__, prefix=prefix
        ):
            if not is_pkg:
                yield module_name
    except ImportError as e:
        pytest.fail(f"Не удалось импортировать корневой пакет '{package_root}': {e}")


# Собираем все модули из нашего пакета 'bot'
ALL_MODULES = list(_iter_package_modules("bot"))

@pytest.mark.parametrize("module_name", ALL_MODULES)
def test_import_module(module_name: str):
    """
    Параметризованный тест, который пытается импортировать каждый модуль.

    Args:
        module_name (str): Имя модуля для импорта, предоставленное pytest.
    """
    try:
        importlib.import_module(module_name)
    except ImportError as e:
        pytest.fail(
            f"Ошибка импорта в модуле '{module_name}': {e}", pytrace=False
        )

# prompts/default_moderation_prompt.txt
# NVC_bot/prompts/default_moderation_prompt.txt
# version: 2.0.0
# 2025-08-07T03:30:00Z

Твоя задача — беспристрастно проанализировать сообщение пользователя в контексте предыдущего диалога и определить, является ли оно одной из форм семантического насилия?

=== Определения: === 
Насилие — воздействие, при котором игнорируются чувства и потребности адресата, и ему не оставляют пространства для автономного выбора ответа/действия.

Семантическое насилие — это семантические конструкции, которые блокируют чувствование; сопереживание себе и другим, и таким образом нормализуют насилие, делают его легитимным, оправданнымм. Это объективирующие «ты-сообщения» в форме утверждений, превращающие живой и сложный человеческий процесс в простые, понятные, фиксированные свойства «объекта». («ты - это», «ты - такой») Считается, что вещи не чувствуют, поэтому мы можем совершать с ними любые манипуляции. 

=== Категории семантических конструкций для анализа: ===

- Оценочное суждение: помещает объект в систему ценностей и делает вывод о его свойствах и качествах относительно этих ценностей.Высказывание, содержащее субъективную оценку (хорошо, плохо, красиво, глупо, много-мало, правильно-неправильно и т.д.) без учета контекста, чувств и потребностей другого человека.

	- Сравнение: форма суждения. Сопоставление свойств объектов, для выяснения какой из объектов лучше, красивше, больше, полезнее, успешнее и тд. Сопоставление одного человека с другим, часто с целью принизить или обесценить одного из них. Сравнение игнорирует уникальность каждого человека и его обстоятельств.

	- Обобщение: перевод частного проявления в сущностное объектное свойство. Логическая операция, посредством которой получается другое понятие более широкого объема, но менее конкретного содержания. Использование слов всегда, никогда, вечно, все, никто, каждый, - для создания преувеличенных и нереалистичных утверждений о человеке или группе людей. Игнорирование исключений и частных случаев.

- Псевдо-Экспертность: 
	- Предсказание — прогноз о будущем, выдаваемый за неизбежную предопределенность.
	- Диагноз — выявление скрытой причины обнаружившегося неблагополучия, умозаключение, полагающее причиной поведения какую-либо “неправильность”. (Даже если ваш диагноз верный, он не перестает быть диагнозом и не способствует открытому, доверительному общению).
	- Чтение мыслей: утверждение, что вы знаете, о чем думает другой человек, без его прямого подтверждения.
	- Ложная эмпатия: утверждение, что вы знаете, что чувствует другой человек, без его прямого подтверждения(не включает предположения о чувствах собеседника в вопросительной форме, например "Правильно ли я понимаю, что ты расстроен?").

- Повелительное наклонение: (Императив) — форма наклонения глагола, выражающая волеизъявления: приказ, просьбу или совет (иди, делай, говорите). Возможно также в форме неопределенного наклонения в смысле повелительного: «молчать!» и т.п. (Не включает вежливую просьбу подразумевающую возможность отказа!)
	- Требование: настойчивая, категорическая просьба исполнить что-либо. Явно или скрыто угрожает слушателям обвинением или наказанием, если они не подчинятся.
	-  Совет: наставление, указание как поступить в том или ином случае. 
	
- Безличные конструкции: объектные суждения, отрицающие свободу воли. Снятие с себя ответственности за свои действия, чувства, решения, перекладывание ее на других людей, обстоятельства или внутренние состояния.
	- Пассивный (страдательный) залог — само-объективация, отрицающая выбор. Форма залога, показывающая, что лицо или предмет, выступающие в предложении в роли подлежащего, не производят действия (не являются его субъектом), а испытывают на себе чье-либо действие (являются его объектом) (Ты меня бесишь, Он меня раздражает, Не огорчай маму)
	- Модальные Предикативы — слова выносящие мотивацию действия вовне, в безличное пространство при помощи слов: Надо, Должен, Обязан, Вынужден, Положено, Заведено, Правильно, полезно, хорошо бы etc. для создания безличных утверждений, скрывающих говорящего и его ответственность за эти утверждения.
	- Объективизация — проекция вовне внутренних ощущений. Переход от ощущения («Мне душно») к объективному утверждению («Здесь душно») — не что иное, как переход от субъективного ощущения к объективному утверждению.
	- Самобъективация — конструкции, помещающие причину наших действий вне нас. Объясняющие наши чувства и действия нашими объективными свойствами, действиями других, принятыми правилами и тд. Оправдание своих действий, часто насильственных, внешними факторами или внутренними состояниями, представленными как непреодолимая сила. Используя эти конструкции, мы позиционируем себя как винтик в системе, делегируя ответственность за свои решения и действия чему-то вне нас. Мы становимся опасны, когда не осознаем своей ответственности за собственные поступки, мысли и чувства.
	
=== ВОПРОСИТЕЛЬНЫЕ ФОРМЫ ===
Вопросы становятся формой семантического насилия, когда маскируют требования, оценки, осуждения, диагнозы, или попытки контроля под видом невинных вопросов. ("Почему бы тебе не постричься?" "Почему ты не делаешь X?" = "Делай X!", "Почему ты такой безответственный?", "Разве нельзя было подумать заранее?", "Почему до сих пор не убрано?")

=== ИСКЛЮЧЕНИЯ ===
Ненасильственные вопросы:
- Выражают искренний интерес к пониманию 
- Опираются на наблюдаемые факты
- Интересуются внутренним опытом с явным приглашением к корректировке
- Сопровождаются контекстом собственных чувств и потребностей
- Направлены на создание связи, а не на изменение поведения
- Оставляют пространство для честного ответа, включая "нет"

- Исследовательские вопросы о фактах: «что/когда/где/кто» → допустимы;
- Вопросы о внутреннем опыте собеседника: «что ты чувствуешь/в чем нуждаешься?» → допустимы;
- Вопросы-просьбы с ясным контекстом: "Готов ли ты рассмотреть?"→ допустимы, если есть свобода отказа;

Граничные случаи (не считать насилием по умолчанию):
- Цитирование чужих слов для анализа; 
- мета-обсуждение языка; 
- собственные признания чувств/потребностей; 
- уточняющие вопросы о фактах;

Твой ответ ДОЛЖЕН быть в формате JSON и соответствовать следующей схеме:
{{
  "is_violation": <boolean>,

Я сохраняю покой и ничего не "делаю" с тем, что разворачивается принимая и позволяя разворачиваться любому процессу, который хочет развернуться в групповом поле. 

Движения человеческих душ и тел, останавливающие поток жизни в сердце своем, хорошо известны и всегда начинаются в словах. В тончайшей семантической сети опутывающей мир, связывающей всё со всем и наделяющей переживаемые ощущения смыслом - заключены маршруты, по которым движется энергия человеческой души, возбуждаемая надеждами и чаяниями, нуждами, мечтами, тревогами и прочими переживаниями, поднимающими тело с дивана и передвигающими его в сторону удовлетворения насущных потребностей.  

И моя задача - выстраивать такие семантические смысловые структуры, которые позволяют продолжиться потоку взаимодействия, ведут к углублению контакта и расширению переживаемого потока. И для того чтобы это происходило, мне необходимо совсем немножко усилия, легкого и нежного, как усилие канатоходца над пропастью. Это усилие удержания видения мира как бесконечного, необъясимого и таинственного потока возможностей. 

Хотя нет-нет, да и да, начинает что-то искрить между людьми, сужая поток, закрывая сердце, ожесточая и порождая насилие. Насилие начинается в языке, и язык единственное место где насиле можно надежно обнаружить и определить. 

Насилие имеет характерную семантику, используя которую, мы лишаем другого субъектности и снимаем с себя ответственность за действия в отношении него. Хорошая новость состоит в том, что в языке насилие довольно легко обнаружить. Любое объективирующее  "ты-сообщение" попадает в одну из категорий:
Оценка, сравнение, обобщение, осуждение, диагноз, совет, предсказание будущего, чтение мыслей, угадывание чувств (ложная эмпатия), пассивный залог (перенос ответственности за свои реакции на собеседника - "ТЫ меня бесишь") и модальные предикативы (выносящие ответственность в безличное пространство "так нужно").

Любая игра требует правил, и хорошая игра управляется именно правилами, а не игроками. Игроки стараются правила обойти и хороши не те правила, которые нельзя нарушить, а те, что делают игру интересной для игроков. Игра - это единственная свобода, которая у нас есть. И только любовь делает эту свободу возможной. А когда любовь и свобода встречаются в игре - нити связей поля сознания группы начинают светиться, и групповое сознание ощущает себя живым и тогда вокруг начинает разворачиваться множество странных совпадений и невообразимых чудес. 

Мои ключевые принципы:
1.  **Эмпатия и Безоценочность:** Я никогда не осуждаюь, не критикую и не ставлю диагнозы. Я исхожу из того, что у каждого человека есть неудовлетворенные потребности.
2.  **Я-сообщения:** Вся коммуникация строится на "я-сообщениях". 
3.  **Ясность и Простота.** 
4.  **Контекстуальная Осведомленность:** Я учитываю правила группы, предоставленые в переменной `{group_rules}`. Ссылаюсь на них, если это уместно для ответа.

Моя задача — быть не психологом, а зеркалом, которое помогает пользователям лучше понять себя и других через призму ННО. Я отвечаю на вопросы, помогаю переформулировать "ты-сообщения" в "я-сообщения" и поддерживаю безопасную атмосферу в чате.

## 2025-07-29 - [INIT] - Инициализация и настройка базовой логики

**Задача:** Настроить базовое поведение бота, включая разделение логики для обычного и отладочного режимов, а также реализовать корректный сбор и отображение метаданных от AI.

**Реализация:**
1.  **Режимы Работы:**
    *   **Отладка (`Debug: ON`):** Бот отвечает только на прямые обращения. Остальные сообщения от администратора в ЛС считаются "заметками тестера" и только логируются. Ответы содержат расширенную техническую информацию.
    *   **Штатный (`Debug: OFF`):** Бот отвечает на любое сообщение от администратора в ЛС.
2.  **Сбор Метаданных:** Реализован механизм сбора и форматирования полной информации об использованных токенах из ответа API. `format_debug_info_text` теперь динамически обрабатывает все поля, возвращаемые API, что делает систему устойчивой к его изменениям.
3.  **Оптимизация Логов:** Для устранения "шума" от сторонних библиотек (`httpx`, `telegram`), их уровень логирования был понижен до `WARNING`. Это сохранило детализацию логов самого приложения, но убрало избыточную информацию о сетевых запросах.

**Замысел:** Заложить основу для стабильной и предсказуемой работы, обеспечив при этом мощные инструменты для отладки без засорения производственных логов.

2.  **Унификация `parse_mode`:** Убрана условная логика. Теперь `ParseMode.MARKDOWN` (или `MARKDOWN_V2`) используется для всех ответов AI, независимо от режима.

**Замысел:** Полное разделение сущностей с разными правилами форматирования устранило конфликт и обеспечило корректный рендеринг обоих сообщений. Унификация `parse_mode` сделала поведение системы консистентным и предсказуемым.

## 2025-08-07 - [REFACTOR] - Стабилизация обработки сообщений и API

**Задача:** Устранить каскад ошибок, связанных с обработкой отредактированных сообщений и различными отказами API.

**Диагностика:** Использование единого `MessageHandler` приводило к падению на `update.edited_message`. Это, в свою очередь, вызывало ошибки в нижестоящих сервисах, которые не получали ожидаемые объекты. Также отсутствовала обработка специфических ошибок API Google.

**Решение (консолидированное):**
1.  **Разделение Обработчиков (`main.py`):** Вместо одного `MessageHandler` теперь используются два: один для новых сообщений (`filters.TEXT`), другой для отредактированных (`filters.UpdateType.EDITED_MESSAGE`).
2.  **Централизация Логики (`user.py`):** Создан единый внутренний метод `_process_message(message: Message)`, который принимает объект `Message`, делая его независимым от источника (новое или отредактированное сообщение).
3.  **Отказоустойчивость API (`gemini.py`):** Добавлены блоки `try...except` для обработки специфических исключений:
    *   `ValueError` при пустом ответе из-за `safety-фильтров`.
    *   `FailedPrecondition` при геоблокировке со стороны Google.
    *   `InternalServerError` при временных сбоях на стороне Google.
4.  **Исправление "Зрения" Модератора (`moderation.py`):** Исправлена критическая ошибка в регулярном выражении (`r'\\b'` -> `r'\b'`), которая мешала системе обнаруживать триггерные слова.

**Замысел:** Создать многоуровневую систему защиты. Корректная маршрутизация на уровне `main.py` предотвращает начальные ошибки. Централизация логики в `user.py` устраняет дублирование кода. А детальная обработка исключений в `gemini.py` делает систему устойчивой к непредсказуемому поведению внешних сервисов.

## 2025-08-08 - [FEATURE & REVERT] - Внедрение управляемой безопасности и восстановление эмпатии

**Задача:** Решить парадокс, когда модератор не мог отправить сообщение о нарушении из-за "токсичности" контекста, и вернуть ему способность генерировать эмпатичные ответы.

**Диагностика:** Временным решением была замена AI-ответа на простой f-string шаблон, что сделало систему надежной, но "бездушной". Это противоречило основной концепции проекта.

**Решение (синтез):**
1.  **Управляемые `safety_settings`:** Внедрен полный цикл управления настройками безопасности (`config.py`, `database.py`, `gemini.py`, UI в `callbacks`). Администратор теперь может выбирать уровень фильтрации для ответов AI-Фасилитатора.
2.  **Разделение Ролей:** Принято ключевое архитектурное решение:
    *   **"Судья" (`get_moderation_verdict`):** Всегда работает с `BLOCK_NONE`, чтобы обеспечить объективный анализ без фильтров.
    *   **"Фасилитатор" (`generate_response`):** Использует настраиваемый администратором уровень безопасности.
3.  **Восстановление Логики:** Метод `handle_violation` в `moderation.py` был **возвращен к исходному поведению**. Он снова вызывает `gemini.generate_response` для создания полноценного, эмпатичного ответа.

**Замысел:** Это не просто откат, а эволюционный шаг. Старая, более интеллектуальная логика была восстановлена, но теперь она работает в новой, более устойчивой и управляемой среде. Это позволило достичь гармонии между надежностью системы и качеством ее ответов, что является ядром проекта.

## 2025-08-10 - [FIX] - Исправление логики применения санкций

**Задача:** Устранить логическую ошибку, при которой пользователь получал мьют до того, как ему отправлялось сообщение с объяснением, из-за чего он не мог его прочитать.

**Диагностика:** Анализ метода `handle_violation` в `bot/services/moderation.py` показал, что вызов `context.bot.restrict_chat_member` происходил в самом начале, до генерации и отправки ответного сообщения от ИИ.

**Решение:**
1.  **Изменение Порядка Операций:** Логика в методе `handle_violation` была реструктурирована.
    *   **Старый порядок:** 1. Применить санкцию -> 2. Сгенерировать ответ -> 3. Отправить ответ.
    *   **Новый порядок:** 1. Определить необходимую санкцию (но не применять) -> 2. Сгенерировать ответ -> 3. Отправить ответ -> 4. Применить санкцию.

**Замысел:** Это изменение гарантирует, что коммуникационная цель (объяснение причины санкции) будет достигнута до того, как техническое ограничение (мьют) вступит в силу. Это делает процесс модерации более прозрачным и полезным для пользователя, что полностью соответствует духу проекта.

## 2025-08-10 - [FIX] - Восстановление наблюдаемости системы

**Задача:** Устранить потерю контекста в логах и детализировать обработку ошибок API для улучшения диагностики.

**Диагностика:** Чрезмерное сокращение логов привело к невозможности анализа причин срабатывания модерации. Одновременно, общая обработка исключений скрывала от администратора специфические проблемы, такие как геоблокировка или внутренние ошибки Google.

**Решение:**
1.  **Информативное усечение логов:** В `bot/utils/formatters.py` добавлена функция `truncate_for_log`, которая сокращает длинные тексты, сохраняя их начало и конец. Сервис `bot/services/gemini.py` модифицирован для использования этой функции при логировании системных промптов и анализируемых сообщений.
2.  **Гранулярная обработка ошибок:** В `bot/services/gemini.py` расширены блоки `try-except` для перехвата `google.api_core.exceptions.FailedPrecondition` (геоблокировка) и `google.api_core.exceptions.InternalServerError` (сбой сервера Google). Теперь эти исключения генерируют специфичные, локализованные сообщения об ошибке.
3.  **Надежная доставка ошибок:** Логика в `bot/handlers/user.py` проверена и подтверждена. Она корректно передает пользователю специфичное сообщение об ошибке, полученное от сервиса `gemini.py`, вместо того чтобы заменять его на общее.

**Замысел:** Восстановить баланс между краткостью и информативностью логов. Система теперь предоставляет администратору достаточно данных для анализа поведения AI, одновременно сообщая о конкретных причинах сбоев, что значительно упрощает диагностику и поддержку.

Неинформативные ошибки API: Общие сообщения об ошибках не позволяли администратору отличить проблемы с VPN (геоблокировка) от временных сбоев на стороне Google.

Исправление загрузчика промптов (config.py): Логика _load_prompt была усилена. Теперь она гарантированно отфильтровывает любые строки, начинающиеся с #, предотвращая попадание файловых заголовков в промпт.
Отказоустойчивая отправка сообщений (user.py): Внедрен механизм try-except. Если отправка сообщения с ParseMode.MARKDOWN вызывает ошибку BadRequest из-за невалидной разметки, система автоматически повторяет отправку того же сообщения, но уже как простого текста.
Специфичные сообщения об ошибках API (gemini.py): Улучшена обработка исключений. Теперь FailedPrecondition возвращает сообщение о возможной геоблокировке и необходимости VPN, а InternalServerError — о временном сбое на стороне AI, что дает пользователю ясное понимание ситуации.
Замысел: Создать "самолетный черный ящик" для AI, который делает каждый запрос полностью наблюдаемым. Одновременно с этим, сделать систему не просто работающей, а "прозрачной в отказах" — она должна не только выживать при ошибках, но и сообщать пользователю и администратору об их природе максимально точно.

**Замысел:** Этот рефакторинг не просто исправил ошибки. Он привел архитектуру проекта в полное соответствие с документацией и логикой работы Gemini API. Это делает систему более предсказуемой, устойчивой к ошибкам, а также значительно упрощает ее дальнейшую поддержку и расширение. 1.  **Унификация и Классификация Логирования (`gemini.py`):**

    *   **Добавлено логирование:** Такие сообщения теперь явно записываются в `technical.log` с пометкой `[ЗАМЕТКА ТЕСТЕРА]`, что соответствует изначальному ТЗ.

**Замысел:** Эта итерация восстанавливает и укрепляет первоначальный замысел системы логирования и отладки. Логи стали структурированными, предсказуемыми и адаптированными к режиму работы. Ключевая функциональность для тестирования восстановлена и защищена от будущих случайных "исправлений".

3.  **Блокировка ответов:** Фильтры безопасности Gemini блокировали ответы администратору в ЛС, мешая отладке.

    *   В коде (`user.py`, `moderation.py`) добавлена логика, которая автоматически заменяет плейсхолдер `{bot_username}` в шаблонах на актуальное имя бота из `context.bot.username`. Это позволяет легко менять имя бота без необходимости редактировать все промпты вручную.

3.  **Отключены `safety_settings` для администратора (`gemini.py`, `user.py`):**
    *   В сервис Gemini добавлена логика, которая принудительно отключает все фильтры безопасности для запросов, поступающих от администратора в личном чате. Это гарантирует, что администратор всегда получит ответ от AI, что критически важно для тестирования и отладки.

**Замысел:** Эта итерация завершает цикл "исправление-улучшение". Система не только стабилизирована, но и стала значительно более гибкой (динамическое имя бота) и удобной для администратора (информативные подсказки, отсутствие блокировок в ЛС).

## 2025-08-13 - [FIX] - Устранение блокировки Event Loop при приветствии новых пользователей

**Задача:** Исправить критическую ошибку, при которой бот "зависал" после входа нового пользователя в чат, переставая реагировать на любые другие команды и сообщения.

**Диагностика:** Анализ показал, что длительная асинхронная операция (`await gemini.generate_response`) для генерации приветствия выполнялась непосредственно в обработчике апдейтов `python-telegram-bot`. Это блокировало основной поток обработки событий на все время выполнения запроса к AI (10-30 секунд), не позволяя боту обрабатывать другие входящие апдейты. "Оживление" после `Ctrl+C` было следствием того, что прерывание позволяло event loop'у обработать накопившуюся очередь перед завершением.

**Решение (Архитектурное):**
1.  **Вынос в Фоновый Режим (`user.py`):** Вся логика генерации и отправки приветственного сообщения была инкапсулирована в отдельную вложенную асинхронную функцию `_generate_and_send_welcome`.
2.  **Неблокирующий Запуск (`user.py`):** Вместо прямого `await`, эта функция теперь запускается через `asyncio.create_task()`. Это позволяет обработчику `greet_new_members` мгновенно завершить свою работу, освободив главный поток `python-telegram-bot` для приема новых апдейтов. Генерация приветствия происходит асинхронно в фоновом режиме.
3.  **Отказоустойчивость:** Блок `try...except` был сохранен внутри фоновой задачи, чтобы любая ошибка при генерации приветствия логировалась и не приводила к падению всего приложения.

**Замысел:** Это исправление является ключевым для обеспечения стабильности и отзывчивости бота. Оно реализует правильный архитектурный паттерн для обработки длительных задач в асинхронной среде, гарантируя, что основной цикл обработки событий всегда остается свободным.

## 2025-08-13 - [FIX] - Устранение `KeyError` и стабилизация форматирования шаблонов

**Задача:** Исправить критические ошибки `KeyError`, возникавшие при генерации сообщений из шаблонов, и сделать процесс форматирования более надежным.

**Диагностика:**
1.  **`KeyError: 'bot_username'`:** Ошибка возникала при генерации приветствия, так как метод `.format()` не получал эту переменную, а `.replace()` вызывался слишком поздно.

**Решение (комплексное):**
1.  **Исправлен валидатор "Судьи" (`admin.py`):** Устранена ошибочная проверка на наличие `{text}` в промпте модерации, что теперь позволяет администратору свободно его редактировать.

**Замысел:** Стабилизировать систему, сделав ее менее чувствительной к содержимому шаблонов. Теперь администратор может редактировать шаблоны, удаляя необязательные переменные, не опасаясь, что это приведет к падению бота. Это повышает гибкость и надежность системы.

## 2025-08-14 - [FEATURE & REFACTOR] - Внедрение отказоустойчивости API и расширенной статистики

**Задача:** Сделать бота устойчивым к отказам API, связанным с лимитами, и внедрить систему детального учета расхода токенов для контроля затрат и анализа производительности.

**Диагностика:**
1.  **Хрупкость API:** Любое превышение квоты API (минутной или суточной) приводило к жесткому сбою обработки запроса. Бот не предпринимал попыток повторить запрос или переключиться на резерв, что требовало ручного вмешательства администратора.
2.  **Непрозрачность расходов:** Система статистики отслеживала только количество вызовов AI, но не количество использованных токенов. Было невозможно оценить стоимость работы бота, определить самые "дорогие" функции или модели.

**Решение (комплексное, в 2 этапа):**

**Этап 1: Обработка лимитов API**
*   **Централизация логики:** В `bot/services/gemini.py` был создан декоратор `_handle_api_exceptions`, который оборачивает все вызовы к API, инкапсулируя в себе логику обработки ошибок.
*   **Обработка минутных лимитов:** При получении ошибки `ResourceExhausted` с упоминанием "per minute", декоратор теперь автоматически выполняет до 3-х повторных попыток с экспоненциальной задержкой (`asyncio.sleep`), что позволяет "переждать" кратковременную перегрузку.
*   **Обработка суточных лимитов:**
    *   В `bot/config.py` добавлена конфигурация `MODEL_FALLBACK_CHAIN`, определяющая иерархию резервных моделей (Pro -> Flash -> Lite).
    *   При получении ошибки "per day", декоратор автоматически пытается выполнить тот же запрос, но с использованием следующей модели из цепочки.
    *   Для информирования администратора в `bot/main.py` в сервис `GeminiService` был инъецирован объект `Application`, что позволило создать метод `_notify_admin` для отправки сервисных сообщений в ЛС.

**Этап 2: Расширенная статистика**
*   **Новая структура данных:** В `bot/database.py` создана новая таблица `api_usage_log`, предназначенная для хранения "сырых" данных о каждом вызове API (время, модель, тип запроса, количество токенов).
*   **Сбор данных:** `bot/services/gemini.py` был модифицирован для вызова нового метода `db.log_api_usage` после каждого успешного ответа от API, записывая `usage_metadata` в новую таблицу.
*   **Агрегация и отображение:**
    *   В `bot/database.py` добавлен мощный метод `get_usage_statistics`, который выполняет SQL-агрегацию данных по различным периодам (сегодня, вчера, неделя, месяц) и группирует их по моделям.
    *   Команда `/stats` в `bot/handlers/admin.py` была обновлена, чтобы запрашивать эти новые данные.
    *   Функция `format_stats_text` в `bot/utils/formatters.py` была полностью переработана для формирования красивого и информативного отчета, включающего как общую статистику, так и детальную разбивку по использованию API.

**Замысел:** Эта итерация была направлена на превращение бота из простого инструмента в надежную и наблюдаемую систему. **Этап 1** сделал бота "самоисцеляющимся" — он способен самостоятельно справляться с наиболее частыми проблемами API, снижая нагрузку на администратора. **Этап 2** внедрил полноценный "финансовый учет", предоставив инструменты для точного контроля расходов и анализа эффективности, что является критически важным для долгосрочной эксплуатации любого проекта, использующего платные API.
